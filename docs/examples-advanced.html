Advanced Features — Examples

Detailed examples for every method and property on Folding, Bookmarks, Inlay Hints, Sticky Scroll, Snippets, Outline, Word Wrap, Change Tracking, and Read-Only Regions.

Folding

Setup and UpdateRegions

using TextAPI.Core.Folding;

var doc     = new TextDocument();
doc.Load("""
class Foo {
    void Bar() {
        int x = 1;
    }
    void Baz() {
        int y = 2;
    }
}
""");

var model = doc.GetFoldingModel();
model.UpdateRegions(new BraceFoldingStrategy(), doc);

Console.WriteLine($"Regions: {model.Regions.Count}");         // 3 (class, Bar, Baz)
Console.WriteLine($"Visible: {model.VisibleLineCount}");       // 9 (all lines)
Console.WriteLine($"IsStale: {model.IsStale}");               // false

Regions — FoldRegion properties

foreach (var region in model.Regions)
{
    Console.WriteLine($"[{region.StartLine}–{region.EndLine}] {region.Label}");
    Console.WriteLine($"  LineCount={region.LineCount} HiddenLineCount={region.HiddenLineCount}");
    Console.WriteLine($"  IsFolded={region.IsFolded}");
}

Fold, Unfold, ToggleFold

int barStartLine = model.Regions[1].StartLine; // "void Bar() {"

model.Fold(barStartLine);
Console.WriteLine(model.VisibleLineCount); // 7  (Bar's body hidden)

model.Unfold(barStartLine);
Console.WriteLine(model.VisibleLineCount); // 9  (restored)

model.ToggleFold(barStartLine);            // fold
Console.WriteLine(model.Regions[1].IsFolded); // true

model.ToggleFold(barStartLine);            // unfold
Console.WriteLine(model.Regions[1].IsFolded); // false

FoldAll / UnfoldAll

model.FoldAll();
Console.WriteLine(model.VisibleLineCount); // 1  (only "class Foo {" visible — outer fold hides everything else)

model.UnfoldAll();
Console.WriteLine(model.VisibleLineCount); // 9

IsLineVisible / IsLineFolded

model.Fold(model.Regions[1].StartLine); // fold Bar

Console.WriteLine(model.IsLineVisible(0));   // true  (class header)
Console.WriteLine(model.IsLineFolded(2));    // true  (Bar body line)
Console.WriteLine(model.IsLineFolded(5));    // false (Baz — not folded)

GetVisibleLines

model.Fold(model.Regions[1].StartLine);
int[] visible = model.GetVisibleLines();
foreach (int line in visible)
    Console.Write($"{line} ");
// 0 1 5 6 7 8  (Bar's lines 2,3,4 are hidden)

ToDisplayLine / ToDocumentLine

model.Fold(model.Regions[1].StartLine);

// Document line 5 (start of Baz) maps to display row 2 (0-based)
int displayRow = model.ToDisplayLine(docLine: 5);
Console.WriteLine(displayRow); // 2

// Display row 2 maps back to document line 5
int docLine = model.ToDocumentLine(displayRow: 2);
Console.WriteLine(docLine); // 5

RegionsChanged and FoldStateChanged events

model.RegionsChanged   += (s, e) => Console.WriteLine("Regions re-detected");
model.FoldStateChanged += (s, e) => Console.WriteLine("Fold state changed");

model.Fold(0);           // fires FoldStateChanged
model.UpdateRegions(new BraceFoldingStrategy(), doc); // fires RegionsChanged

Bookmarks

Toggle and IsBookmarked

using TextAPI.Core.Navigation;

doc.Load("line0\nline1\nline2\nline3");
var bm = doc.GetBookmarkModel();

bool added   = bm.Toggle(1); Console.WriteLine(added);             // true  (bookmarked)
bool removed = bm.Toggle(1); Console.WriteLine(removed);           // false (removed)

bm.Toggle(0);
bm.Toggle(2);
Console.WriteLine(bm.IsBookmarked(0)); // true
Console.WriteLine(bm.IsBookmarked(1)); // false  (removed above)
Console.WriteLine(bm.IsBookmarked(2)); // true

GetAll

bm.Toggle(3);
foreach (int line in bm.GetAll())
    Console.Write($"{line} ");
// 0 2 3  (ascending order)

NextBookmark / PrevBookmark

// From line 1, find the next bookmark
int? next = bm.NextBookmark(fromLine: 1);
Console.WriteLine(next); // 2

// From line 2, find the previous
int? prev = bm.PrevBookmark(fromLine: 2);
Console.WriteLine(prev); // 0

// No bookmark found → null
int? none = bm.NextBookmark(fromLine: 10);
Console.WriteLine(none.HasValue); // false

Navigation with cursor

var cursor = new TextCursor(doc);
int? nextLine = bm.NextBookmark(cursor.CaretLine);
if (nextLine.HasValue)
    cursor.MoveTo(doc.PositionToOffset(nextLine.Value, 0));

OnInsert / OnDelete — remap after edits

// Bookmarks at lines 0, 2, 3
// Insert 2 new lines before line 1 → bookmarks at 0, 2, 3 become 0, 4, 5
bm.OnInsert(atLine: 1, count: 2);
Console.WriteLine(bm.GetAll().SequenceEqual(new[] { 0, 4, 5 })); // true

// Delete line 1 → bookmarks at 0, 4, 5 → after delete become 0, 3, 4
bm.OnDelete(atLine: 1, count: 1);
Console.WriteLine(bm.GetAll().SequenceEqual(new[] { 0, 3, 4 })); // true

Clear

bm.Clear();
Console.WriteLine(bm.GetAll().Any()); // false

BookmarksChanged event

bm.BookmarksChanged += (s, e) => Console.WriteLine("Bookmarks updated — refresh gutter");
bm.Toggle(5); // fires the event

Inlay Hints

AddHint

using TextAPI.Core.InlayHints;

doc.Load("Foo(42, true, \"hello\")");
var hints = doc.GetInlayHintModel();

Guid id1 = hints.AddHint(new InlayHint(
    offset:  4,
    text:    "value:",
    kind:    InlayHintKind.Parameter,
    tooltip: "int value"
));
Guid id2 = hints.AddHint(new InlayHint(
    offset:  8,
    text:    "flag:",
    kind:    InlayHintKind.Parameter,
    tooltip: "bool flag"
));

Console.WriteLine(hints.AllHints.Count); // 2

InlayHint properties

var h = hints.AllHints[0];
Console.WriteLine(h.Id);       // Guid
Console.WriteLine(h.Offset);   // 4
Console.WriteLine(h.Text);     // "value:"
Console.WriteLine(h.Kind);     // Parameter
Console.WriteLine(h.Tooltip);  // "int value"

GetHintsInRange

var visible = hints.GetHintsInRange(0, 10).ToList();
Console.WriteLine(visible.Count); // 2  (both hints in range)

GetHintsByKind

hints.AddHint(new InlayHint(15, "→ string", InlayHintKind.Type));

var paramHints = hints.GetHintsByKind(InlayHintKind.Parameter).ToList();
var typeHints  = hints.GetHintsByKind(InlayHintKind.Type).ToList();
Console.WriteLine(paramHints.Count); // 2
Console.WriteLine(typeHints.Count);  // 1

GetHintAt

InlayHint? h = hints.GetHintAt(offset: 4);
Console.WriteLine(h?.Text); // "value:"

InlayHint? none = hints.GetHintAt(offset: 99);
Console.WriteLine(none.HasValue); // false

SetHints — replace all at once

// Typically called when language server provides fresh hints
hints.SetHints(new[]
{
    new InlayHint(0, "x:", InlayHintKind.Parameter),
    new InlayHint(5, "y:", InlayHintKind.Parameter)
});
Console.WriteLine(hints.AllHints.Count); // 2  (previous hints replaced)

RemoveHint and ClearHints

hints.RemoveHint(id1);
Console.WriteLine(hints.AllHints.Count); // 1

hints.ClearHints();
Console.WriteLine(hints.AllHints.Count); // 0

HintsChanged event

hints.HintsChanged += (s, e) => Console.WriteLine("Hints updated — invalidate viewport");
hints.AddHint(new InlayHint(0, "test:", InlayHintKind.Other)); // fires

Sticky Scroll

using TextAPI.Core.StickyScroll;

// Requires a FoldingModel with regions detected
var foldingModel = doc.GetFoldingModel();
foldingModel.UpdateRegions(new BraceFoldingStrategy(), doc);

// Called on every scroll event — firstVisibleLine is the viewport's top line (0-based)
var context = StickyScroll.GetContext(foldingModel, firstVisibleLine: 3);

Console.WriteLine($"Sticky headers: {context.Count}");
foreach (StickyScrollEntry entry in context)
{
    Console.WriteLine($"  Line {entry.DocumentLine}: {entry.Label}");
}
// e.g.:
//   Line 0: class Foo {
//   Line 1:     void Bar() {

// StickyScrollEntry is a record struct — destructure if needed
var (label, docLine) = context[0];
Console.WriteLine($"{label} @ {docLine}");

Snippets

Parse — parse once

using TextAPI.Core.Snippets;

Snippet template = SnippetEngine.Parse("for (int $1 = 0; $1 < $2; $1++) {\n    $0\n}");

Console.WriteLine(template.TabStopIndices.SequenceEqual(new[] { 1, 2 })); // true
Console.WriteLine(template.HasExitStop);  // true  ($0 is the exit stop)

BeginSnippet

doc.Load("    ");  // 4 spaces cursor position
SnippetSession session = SnippetEngine.BeginSnippet(doc, template, insertOffset: 4);

Console.WriteLine(session.IsActive);            // true
Console.WriteLine(session.CurrentTabStopIndex); // -1 (before first NextTabStop call)

NextTabStop — Tab key handler

var cursor = new TextCursor(doc);

TabStop? stop1 = session.NextTabStop();  // move to $1
if (stop1 is not null)
{
    cursor.SetSelection(stop1.Value.Offset, stop1.Value.Offset + stop1.Value.Length);
    Console.WriteLine($"TabStop {stop1.Value.Index} at {stop1.Value.Offset}+{stop1.Value.Length}");
}

TabStop properties

var stop = session.NextTabStop()!.Value;
Console.WriteLine(stop.Index);   // 1, 2, or 0 (exit)
Console.WriteLine(stop.Offset);  // current offset in document
Console.WriteLine(stop.Length);  // placeholder length

UpdateTabStop — update mirrors

// User typed "i" for the loop variable — update $1 and all mirrors
session.UpdateTabStop(tabStopIndex: 1, newText: "i");
// All $1 placeholders in the snippet text are now "i"

PrevTabStop — Shift+Tab

TabStop? prev = session.PrevTabStop();
if (prev is not null)
    cursor.SetSelection(prev.Value.Offset, prev.Value.Offset + prev.Value.Length);

GetTabStops and GetPrimary

// Get all tab-stop positions for index 1 (including mirrors)
var all1 = session.GetTabStops(1);
Console.WriteLine(all1.Count); // e.g. 3 for "for (int $1 = 0; $1 < $2; $1++)"

// Primary tab stop for index 1
var primary = session.GetPrimary(1);
Console.WriteLine(primary.HasValue); // true

Commit and Cancel

// After last tab stop → commit (keep inserted text)
TabStop? exitStop = session.NextTabStop(); // reaches $0
if (exitStop == null)
    session.Commit();  // finalize — IsActive becomes false

Console.WriteLine(session.IsActive);  // false

// Or cancel (undo the entire snippet insertion):
// session.Cancel();
// Console.WriteLine(session.IsActive);  // false

BeginSnippet with clipboard text and filename

Snippet t2 = SnippetEngine.Parse("// File: $TM_FILENAME\n$CLIPBOARD\n$0");
var s2 = SnippetEngine.BeginSnippet(
    doc, t2, insertOffset: 0,
    clipboardText: "pasted content",
    filename:      "MyFile.cs"
);

Outline

using TextAPI.Core.Outline;

var folding = doc.GetFoldingModel();
folding.UpdateRegions(new BraceFoldingStrategy(), doc);

IReadOnlyList<OutlineNode> outline = OutlineProvider.GetOutline(folding);

// Recursive tree printer
void Print(IReadOnlyList<OutlineNode> nodes, int indent = 0)
{
    foreach (var n in nodes)
    {
        Console.WriteLine($"{new string(' ', indent * 2)}{n.Label}");
        Console.WriteLine($"{new string(' ', indent * 2)}  lines {n.StartLine}–{n.EndLine}, depth {n.Depth}");
        Print(n.Children, indent + 1);
    }
}
Print(outline);
// class Foo {
//   lines 0–8, depth 0
//   void Bar() {
//     lines 1–3, depth 1
//   void Baz() {
//     lines 4–6, depth 1

OutlineNode — all properties

var first = outline[0];
Console.WriteLine(first.Label);                     // region header text
Console.WriteLine(first.StartLine);                  // 0-based doc line
Console.WriteLine(first.EndLine);                    // 0-based doc line
Console.WriteLine(first.Depth);                      // 0 (top-level)
Console.WriteLine(first.Children.Count);             // 2 (Bar + Baz)

// Navigate to a node in the editor
var cursor = new TextCursor(doc);
cursor.MoveTo(doc.PositionToOffset(first.StartLine, 0));

Word Wrap

Setup

using TextAPI.Core.WordWrap;

doc.Load("Short line\nThis is a much longer line that will certainly wrap at 20 columns\nEnd");
var wrap = doc.GetWordWrapModel(viewportWidth: 20);

Console.WriteLine($"ViewportWidth:  {wrap.ViewportWidth}");  // 20
Console.WriteLine($"DisplayRowCount: {wrap.DisplayRowCount}"); // more than 3 (long line wraps)

ToDisplayRow

// Line 0 ("Short line") → display row 0
Console.WriteLine(wrap.ToDisplayRow(0)); // 0

// Line 1 (long line) → display row 1
Console.WriteLine(wrap.ToDisplayRow(1)); // 1

// Line 2 ("End") → display row beyond the long line's rows
Console.WriteLine(wrap.ToDisplayRow(2)); // e.g. 5

ToDocumentLine

// Display row 0 → document line 0
Console.WriteLine(wrap.ToDocumentLine(0)); // 0

// Display rows inside the long line all map to line 1
Console.WriteLine(wrap.ToDocumentLine(1)); // 1
Console.WriteLine(wrap.ToDocumentLine(2)); // 1
Console.WriteLine(wrap.ToDocumentLine(3)); // 1

WrappedRowCount and IsWrapped

Console.WriteLine(wrap.WrappedRowCount(0)); // 1  (short line — one row)
Console.WriteLine(wrap.WrappedRowCount(1)); // e.g. 4 (long line wraps to 4 rows)

Console.WriteLine(wrap.IsWrapped(0)); // false
Console.WriteLine(wrap.IsWrapped(1)); // true

GetWrappedSegments

// Character offset ranges for each visual row of line 1
var segs = wrap.GetWrappedSegments(1);
foreach (var (start, end) in segs)
    Console.WriteLine($"  row chars [{start},{end}): \"{doc.GetText(start, end - start)}\"");

Resize

wrap.Resize(newWidth: 80);
Console.WriteLine(wrap.ViewportWidth);  // 80
Console.WriteLine(wrap.IsWrapped(1));    // false  (80 cols, line fits)

wrap.WrapChanged += (s, e) => Console.WriteLine("Wrap layout changed — redraw viewport");

Change Tracking

Setup and SetBaseline

using TextAPI.Core.ChangeTracking;

doc.Load("line0\nline1\nline2\nline3");
var tracker = doc.GetChangeTracker();
tracker.SetBaseline();

Console.WriteLine(tracker.HasAnyChanges); // false

GetStatus — per-line change marker

// Edit line 1
doc.Replace(doc.PositionToOffset(1, 0), doc.GetLine(1).Length, "MODIFIED");
// Insert a new line after line 2
doc.Insert(doc.PositionToOffset(3, 0), "new line\n");

Console.WriteLine(tracker.GetStatus(0)); // Clean
Console.WriteLine(tracker.GetStatus(1)); // Modified
Console.WriteLine(tracker.GetStatus(2)); // Clean
Console.WriteLine(tracker.GetStatus(3)); // Added
Console.WriteLine(tracker.GetStatus(4)); // Clean  (old line3)

HasAnyChanges

Console.WriteLine(tracker.HasAnyChanges); // true

ChangedLines

foreach (var (lineIndex, status) in tracker.ChangedLines())
    Console.WriteLine($"Line {lineIndex}: {status}");
// Line 1: Modified
// Line 3: Added

HasDeletionAbove

// Delete line 2 from the baseline (from the perspective of the current document)
doc.Delete(doc.PositionToOffset(2, 0), doc.GetLine(2).Length + 1);

// HasDeletionAbove: true if baseline had lines deleted above this doc line
Console.WriteLine(tracker.HasDeletionAbove(2)); // true  (a line was removed)
Console.WriteLine(tracker.HasDeletionAbove(0)); // false

DeletionPoints

foreach (int point in tracker.DeletionPoints())
    Console.WriteLine($"Deletion marker above line {point}");

SetBaseline after save

async Task SaveAsync()
{
    await doc.SaveFileAsync();
    tracker.SetBaseline();  // capture saved state as new baseline
    Console.WriteLine(tracker.HasAnyChanges); // false
}

ChangesUpdated event

tracker.ChangesUpdated += (s, e) =>
{
    // Re-render change gutter
    for (int i = 0; i < doc.LineCount; i++)
    {
        char marker = tracker.GetStatus(i) switch
        {
            LineStatus.Added    => '+',
            LineStatus.Modified => '~',
            _                    => ' '
        };
        Console.Write(marker);
    }
};

Read-Only Regions

Protect and IsReadOnly

using TextAPI.Core.ReadOnly;

doc.Load("/* PROTECTED */\nEditable content\n/* END */");
var readOnly = doc.GetReadOnlyModel();

// Protect the header (chars 0–15)
Guid id = readOnly.Protect(start: 0, end: 15);

Console.WriteLine(readOnly.IsReadOnly(0));   // true
Console.WriteLine(readOnly.IsReadOnly(5));   // true
Console.WriteLine(readOnly.IsReadOnly(16));  // false  (outside protected range)

IsRangeReadOnly

Console.WriteLine(readOnly.IsRangeReadOnly(0, 5));   // true   (overlaps [0,15))
Console.WriteLine(readOnly.IsRangeReadOnly(10, 3)); // true   (inside [0,15))
Console.WriteLine(readOnly.IsRangeReadOnly(16, 5)); // false  (no overlap)

ReadOnlyViolationException

doc.EnforceReadOnly = true; // default

try
{
    doc.Insert(5, "HACK");  // inside protected region
}
catch (ReadOnlyViolationException ex)
{
    Console.WriteLine(ex.Message); // e.g. "Offset 5 is within a read-only region"
}

Bypass protection programmatically

doc.EnforceReadOnly = false;
doc.Insert(0, "ALLOWED"); // succeeds
doc.EnforceReadOnly = true;

GetRegions

readOnly.Protect(20, 30);
foreach (var (rid, start, end) in readOnly.GetRegions())
    Console.WriteLine($"[{start},{end})");
// [0,15)
// [20,30)

Unprotect / UnprotectAll

readOnly.Unprotect(id);  // remove [0,15) protection
Console.WriteLine(readOnly.GetRegions().Count()); // 1

readOnly.UnprotectAll();
Console.WriteLine(readOnly.GetRegions().Count()); // 0

RegionsChanged event

readOnly.RegionsChanged += (s, e) => Console.WriteLine("Protected regions changed");
readOnly.Protect(0, 5); // fires event
