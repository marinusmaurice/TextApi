TextDocument — Examples

A focused example for every constructor parameter, property, and method on TextDocument.

Construction

Default constructor

var doc = new TextDocument();
// Empty document, plain-text tokeniser, 1 000-step undo history
Console.WriteLine(doc.Length);    // 0
Console.WriteLine(doc.LineCount); // 1 (an empty document still has 1 line)

Custom undo-history limit

// Keep only the last 50 undo steps (saves memory in long-running editors)
var doc = new TextDocument(undoHistoryLimit: 50);

Custom syntax tokeniser

// Plug in your own tokeniser — affects LanguageId and GetSyntaxTokens()
var doc = new TextDocument(myTokeniser);

Load

Load(string content)

var doc = new TextDocument();
doc.Load("Hello\nWorld\n");
Console.WriteLine(doc.LineCount);  // 3  (2 lines + trailing empty line)
Console.WriteLine(doc.IsModified); // false — Load resets IsModified

Load with FilePath

// Provide a logical path (does NOT read from disk — content is the string)
doc.Load("int x = 1;", filePath: "/virtual/snippet.cs");
Console.WriteLine(doc.FilePath); // "/virtual/snippet.cs"

Load resets undo history

doc.Load("line1\nline2");
doc.Insert(0, "prefix ");
Console.WriteLine(doc.CanUndo); // true

doc.Load("fresh content");        // ← resets everything
Console.WriteLine(doc.CanUndo);   // false — undo stack cleared

Load(null) throws ArgumentNullException

try { doc.Load(null!); }
catch (ArgumentNullException ex) { Console.WriteLine(ex.ParamName); } // "content"

LoadFileAsync — auto-detect encoding and EOL

await doc.LoadFileAsync("README.md");
Console.WriteLine(doc.OriginalEolStyle);               // LF / CRLF / CR
Console.WriteLine(doc.DetectedEncoding?.Encoding.WebName); // "utf-8"
Console.WriteLine(doc.HasBom);                          // true/false

LoadFileAsync — override encoding

// Force Latin-1 regardless of detection
await doc.LoadFileAsync("legacy.txt", encoding: System.Text.Encoding.Latin1);

Properties

Length and LineCount

doc.Load("abc\nde\nf");
Console.WriteLine(doc.Length);    // 8  (a b c \n d e \n f)
Console.WriteLine(doc.LineCount); // 3

IsModified

doc.Load("Hello");
Console.WriteLine(doc.IsModified); // false
doc.Insert(5, "!");
Console.WriteLine(doc.IsModified); // true
await doc.SaveFileAsync("out.txt");
Console.WriteLine(doc.IsModified);   // false — cleared on successful save

FilePath

await doc.LoadFileAsync("/data/file.txt");
Console.WriteLine(doc.FilePath); // "/data/file.txt"

// After Load(string), FilePath is whatever you passed (or null)
doc.Load("hi");
Console.WriteLine(doc.FilePath); // null

doc.Load("hi", filePath: "/virtual/x.cs");
Console.WriteLine(doc.FilePath); // "/virtual/x.cs"

LanguageId

// Comes from the active tokeniser
Console.WriteLine(doc.LanguageId); // "plaintext" (default) or "csharp", "json", etc.

OriginalEolStyle and SaveEolStyle

await doc.LoadFileAsync("windows.txt");
Console.WriteLine(doc.OriginalEolStyle); // CRLF

// Override EOL on save
doc.SaveEolStyle = EolStyle.LF;
await doc.SaveFileAsync();           // saves with \n line endings

EnforceReadOnly

// Default: true — edits in protected ranges throw ReadOnlyViolationException
// Disable to allow programmatic bypass
doc.EnforceReadOnly = false;
doc.Insert(0, "bypassed"); // works even in protected region
doc.EnforceReadOnly = true;

PieceCount (diagnostic)

Console.WriteLine(doc.PieceCount); // pieces in piece table — grows with edits
doc.Compact();                      // collapses pieces
Console.WriteLine(doc.PieceCount);  // much lower after compaction

Reading Text

GetText() — full document

doc.Load("line1\r\nline2\r\n");
Console.WriteLine(doc.GetText().Replace("\n", "↵")); // "line1↵line2↵"
// GetText() always normalises to \n internally

GetText(offset, length) — substring

doc.Load("Hello, World!");
Console.WriteLine(doc.GetText(7, 5)); // "World"

GetTextWithEol() — preserves EOL style

doc.SaveEolStyle = EolStyle.CRLF;
string windows = doc.GetTextWithEol(); // uses \r\n

GetLine(lineIndex)

doc.Load("first\nsecond\nthird");
Console.WriteLine(doc.GetLine(0)); // "first"    (no \n)
Console.WriteLine(doc.GetLine(1)); // "second"
Console.WriteLine(doc.GetLine(2)); // "third"

Position Conversion

OffsetToPosition

doc.Load("abc\nde\nfg");
var (line, col) = doc.OffsetToPosition(5);
Console.WriteLine($"line={line} col={col}"); // line=1 col=1  (0-based)
// offset 0='a',1='b',2='c',3='\n',4='d',5='e' → line 1, col 1

PositionToOffset

doc.Load("abc\nde\nfg");
int off = doc.PositionToOffset(2, 1); // line 2, col 1 → offset 8
Console.WriteLine(off);             // 8
Console.WriteLine(doc.GetText(8, 1)); // "g"

Round-trip

for (int i = 0; i < doc.Length; i++)
{
    var (l, c) = doc.OffsetToPosition(i);
    int back   = doc.PositionToOffset(l, c);
    if (back != i) throw new Exception($"Mismatch at {i}");
}

Mutations

Insert

doc.Load("Hello World");
doc.Insert(5, ",");
Console.WriteLine(doc.GetText()); // "Hello, World"

Insert at offset 0 (prepend)

doc.Load("World");
doc.Insert(0, "Hello, ");
Console.WriteLine(doc.GetText()); // "Hello, World"

Insert at end (append)

doc.Load("Hello");
doc.Insert(doc.Length, "!");
Console.WriteLine(doc.GetText()); // "Hello!"

Insert(null) throws

try { doc.Insert(0, null!); }
catch (ArgumentNullException ex) { Console.WriteLine(ex.ParamName); } // "text"

Delete

doc.Load("Hello, World!");
doc.Delete(5, 2); // delete ", " at offset 5
Console.WriteLine(doc.GetText()); // "HelloWorld!"

Delete entire content

doc.Delete(0, doc.Length);
Console.WriteLine(doc.Length); // 0

Replace

doc.Load("Hello World");
doc.Replace(6, 5, "TextAPI"); // delete 5 chars at 6, insert "TextAPI"
Console.WriteLine(doc.GetText()); // "Hello TextAPI"

ReplaceAll

doc.Load("foo bar foo baz foo");
int count = doc.ReplaceAll("foo", "qux");
Console.WriteLine(count);            // 3
Console.WriteLine(doc.GetText()); // "qux bar qux baz qux"

ReplaceAll — case-insensitive

var opts = new SearchOptions { CaseSensitive = false };
doc.Load("TODO todo Todo");
int n = doc.ReplaceAll("todo", "DONE", opts);
Console.WriteLine(n);            // 3
Console.WriteLine(doc.GetText()); // "DONE DONE DONE"

ReplaceAll — whole word

var opts = new SearchOptions { WholeWord = true };
doc.Load("cat concatenate cats");
int n = doc.ReplaceAll("cat", "dog", opts);
Console.WriteLine(n);            // 1  (only the standalone "cat")
Console.WriteLine(doc.GetText()); // "dog concatenate cats"

ExecuteComposite — multi-edit as one undo step

using TextAPI.Core.Commands;

doc.Load("Hello World");
doc.ExecuteComposite("Rename", new IEditorCommand[]
{
    new InsertCommand(0,  "// Top\n"),
    new InsertCommand(100, "\n// Bottom"),
});
// One Undo() removes both inserts at once
doc.Undo();  // doc.GetText() == "Hello World"

Undo / Redo

CanUndo, Undo, CanRedo, Redo

doc.Load("A");
doc.Insert(1, "B");
doc.Insert(2, "C");

Console.WriteLine(doc.GetText()); // "ABC"
Console.WriteLine(doc.CanUndo);  // true

doc.Undo();
Console.WriteLine(doc.GetText()); // "AB" (BC were coalesced — one undo step)
Console.WriteLine(doc.CanRedo);  // true

doc.Redo();
Console.WriteLine(doc.GetText()); // "ABC"

UndoDescriptions and RedoDescriptions

doc.Load("x");
doc.Insert(0, "hello");
doc.ReplaceAll("x", "y");

foreach (var d in doc.UndoDescriptions)
    Console.WriteLine(d);
// ReplaceAll ("x" → "y", 1 replacements)
// Insert

FlushUndoGroup — seal coalescing group

doc.Load("");
doc.Insert(0, "a");
doc.Insert(1, "b");
doc.FlushUndoGroup();   // seal — "ab" is now one step
doc.Insert(2, "c");      // new step

doc.Undo();
Console.WriteLine(doc.GetText()); // "ab"  (only "c" undone)
doc.Undo();
Console.WriteLine(doc.GetText()); // ""    ("ab" undone)

Search

FindAll

doc.Load("cat bat cat rat cat");
foreach (var m in doc.FindAll("cat"))
    Console.WriteLine($"{m.Offset}+{m.Length}");
// 0+3
// 8+3
// 16+3

FindFirst

doc.Load("one two one");
var m = doc.FindFirst("one");
Console.WriteLine(m?.Offset); // 0

FindNext — forward scan from cursor

doc.Load("one two one three one");
int from = 4;  // skip first "one"
var m = doc.FindNext("one", from);
Console.WriteLine(m?.Offset); // 8

FindPrev — backward scan

doc.Load("one two one three");
var m = doc.FindPrev("one", doc.Length);
Console.WriteLine(m?.Offset); // 8

FindNext with wrap-around

int caretOffset = 15;
SearchMatch? hit = doc.FindNext("one", caretOffset + 1)
                ?? doc.FindNext("one", 0); // wrap
Console.WriteLine(hit?.Offset);

CountMatches

doc.Load("TODO fix this TODO and that TODO");
Console.WriteLine(doc.CountMatches("TODO")); // 3

FindAll with regex

var opts = new SearchOptions { UseRegex = true };
doc.Load("2024-01-15 and 2025-06-30");
var dates = doc.FindAll(@"\d{4}-\d{2}-\d{2}", opts).ToList();
Console.WriteLine(dates.Count); // 2
foreach (var d in dates)
    Console.WriteLine(doc.GetText(d.Offset, d.Length));
// 2024-01-15
// 2025-06-30

Saving

SaveFileAsync — preserve encoding/EOL

await doc.LoadFileAsync("file.txt");
doc.ReplaceAll("TODO", "DONE");
await doc.SaveFileAsync(); // saves to same path, same encoding, same EOL

SaveFileAsync — explicit path

await doc.SaveFileAsync("output/file.txt");

SaveFileAsync — override encoding

doc.SaveEncoding = System.Text.Encoding.UTF8;
await doc.SaveFileAsync();
// SaveEncoding remains set for future saves; set to null to revert to original

SaveAsync — to stream

using var ms = new MemoryStream();
await doc.SaveAsync(ms);
byte[] bytes = ms.ToArray();

SaveAsync — override encoding for this save only

await doc.SaveAsync(stream, encoding: System.Text.Encoding.UTF8);

Statistics

GetStats()

doc.Load("Hello 👨‍👩‍👧‍👦 World\nLine 2");
var s = doc.GetStats();

Console.WriteLine(s.GraphemeCount);  // user-perceived chars (emoji = 1)
Console.WriteLine(s.CodeUnitCount);  // same as doc.Length (UTF-16)
Console.WriteLine(s.RuneCount);      // Unicode code points
Console.WriteLine(s.WordCount);      // whitespace-delimited words
Console.WriteLine(s.LineCount);      // same as doc.LineCount
Console.WriteLine(s.DisplayColumns); // East-Asian width columns (CJK=2, emoji=2, ASCII=1)

Compact

// After thousands of edits the piece table can fragment.
// Compact() merges all pieces into one, restoring O(1) access.
Console.WriteLine(doc.PieceCount); // e.g. 8 432
doc.Compact();
Console.WriteLine(doc.PieceCount); // 1

Syntax Tokenisation

GetSyntaxTokens — incremental (cached)

// Use GetSyntaxTokens for rendering — hits the incremental cache
var tokens = doc.GetSyntaxTokens(0);
foreach (var t in tokens)
    Console.WriteLine($"{t.Kind} [{t.Start},{t.End})");

SetTokeniser — switch language

doc.SetTokeniser(myCSharpTokeniser);
Console.WriteLine(doc.LanguageId); // "csharp"
// Highlight cache is fully invalidated

InvalidateHighlightCache

// Force re-highlight from line 10 onwards (e.g. after a theme change)
doc.InvalidateHighlightCache(fromLine: 10);

Bracket Matching & Auto-Indent

FindMatchingBracket

doc.Load("if (x) { return; }");
int open  = 7;  // offset of '{'
int close = doc.FindMatchingBracket(open);
Console.WriteLine(close); // 18  (offset of '}')

// Returns -1 if not a bracket or unmatched
Console.WriteLine(doc.FindMatchingBracket(3)); // 5  ('(' ↔ ')')

GetAutoIndent

doc.Load("function foo() {\n    // body\n}");
// Caret is at end of "function foo() {" → new line should be indented one level
string indent = doc.GetAutoIndent(caretOffset: 17);
Console.WriteLine(indent.Replace(" ", "·")); // "····" (4 spaces)

GetClosingBraceIndent

// User types '}' — tell the editor what indent to use
string? dedent = doc.GetClosingBraceIndent(caretOffset: cursor.CaretOffset);
if (dedent is not null)
    // replace leading whitespace of current line with dedent;

// Each model is created once on first access and reused
var bookmarks    = doc.GetBookmarkModel();

// Cursor position history
var history = doc.GetCursorHistory();
doc.GoTo(line: 5, column: 0); // pushes into history
