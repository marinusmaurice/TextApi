Batch Rename Across Files

Load each file, run a pipeline, save only if something changed:

using TextAPI.Core;
using TextAPI.Operations;
using TextAPI.Operations.Pipeline;

async Task RenameAsync(string[] files, string oldName, string newName)
{
    var doc = new TextDocument();

    foreach (var path in files)
    {
        await doc.LoadFileAsync(path);

        var result = new DocumentPipeline(doc)
            .Add(new ReplaceAllOperation { Find = oldName, Replace = newName })
            .Execute();

        if (result.Success && doc.IsModified)
        {
            await doc.SaveFileAsync();
            Console.WriteLine($"Updated: {path}");
        }
    }
}

Add / Replace License Headers

Prepend a license header to every source file, replacing any existing one:

const string OLD_MARKER = "// Copyright";
const string NEW_HEADER = """
// Copyright 2025 Acme Corp. All rights reserved.
// Licensed under the MIT License.

""";

async Task ApplyHeaderAsync(string path)
{
    var doc = new TextDocument();
    await doc.LoadFileAsync(path);

    PipelineResult result;

    var existing = doc.FindNext(OLD_MARKER, 0);
    if (existing is not null)
    {
        // Replace up to and including the blank line after the header
        var (headerLine, _) = doc.OffsetToPosition(existing.Value.Offset);
        int headerEnd = doc.PositionToOffset(headerLine + 2, 0);
        result = new DocumentPipeline(doc)
            .Add(new DeleteAtOperation(existing.Value.Offset, headerEnd - existing.Value.Offset))
            .Add(new InsertAtOperation(0, NEW_HEADER))
            .Execute();
    }
    else
    {
        result = new DocumentPipeline(doc)
            .Add(new PrependOperation { Text = NEW_HEADER })
            .Execute();
    }

    if (result.Success) await doc.SaveFileAsync();
    else Console.WriteLine($"Failed: {result.ErrorMessage}");
}

Normalize a File (EOL, Whitespace, Encoding)

var doc = new TextDocument();
await doc.LoadFileAsync("legacy.txt");

// Normalize content
var result = new DocumentPipeline(doc)
    .Add(new TrimTrailingWhitespaceOperation())
    .Add(new EnsureTrailingNewlineOperation())
    .Execute();

// Force UTF-8 LF on save
doc.SaveEncoding = System.Text.Encoding.UTF8;
doc.SaveEolStyle  = EolStyle.LF;

if (result.Success)
    await doc.SaveFileAsync();

Console.WriteLine(result.AuditLog.Summary);

AI-Generated Edits via JSON Pipeline

Feed LLM output directly to OperationMapper — no parsing glue needed:

using TextAPI.Operations.Serialization;

// JSON produced by an LLM (tool call response)
string llmJson = """
{
  "operations": [
    { "type": "REPLACE_ALL", "find": "var ",    "replace": "let " },
    { "type": "REPLACE_ALL", "find": "Console.WriteLine", "replace": "console.log" },
    { "type": "TRIM_TRAILING_WHITESPACE" },
    { "type": "ENSURE_TRAILING_NEWLINE" }
  ]
}
""";

var ops    = OperationMapper.FromPipelineJson(llmJson);
var result = new DocumentPipeline(doc).Add(ops).Execute();

if (!result.Success)
{
    // Send error back to LLM for retry
    string feedback = $"Op #{result.FailedOperationIndex} failed: {result.ErrorMessage}";
    // ... retry logic ...
}
else
{
    Console.WriteLine(result.AuditLog.Summary);
}

Diagnostic Pass with Decorations

Simulate a language server: find patterns and annotate them:

using TextAPI.Core.Decorations;

// Clear previous diagnostics
doc.Decorations.RemoveAllOfType(DecorationType.ErrorSquiggle);
doc.Decorations.RemoveAllOfType(DecorationType.WarningSquiggle);

// Mark all TODO comments as warnings
var opts = new SearchOptions { UseRegex = true };
foreach (var m in doc.FindAll(@"//\s*TODO\b.*", opts))
{
    doc.AddDecoration(m.Offset, m.Offset + m.Length,
        DecorationType.WarningSquiggle, tag: "TODO");
}

// Count and report
int warnings = doc.Decorations.GetDecorationsInRange(0, doc.Length)
    .Count(d => d.Type == DecorationType.WarningSquiggle);
Console.WriteLine($"{warnings} TODO warning(s)");

Generate a Diff Report

using TextAPI.Core.Diff;

var original = File.ReadAllText("original.cs");
var modified = File.ReadAllText("modified.cs");

var diff = TextDiff.Diff(original, modified);

Console.WriteLine($"Changes: +{diff.AddedLines} / -{diff.DeletedLines}");
Console.WriteLine();
Console.WriteLine(diff.ToUnifiedDiff("original.cs", "modified.cs"));

// Also render inline char diffs for each changed hunk
foreach (var hunk in diff.Hunks.Where(h => h.Kind != DiffKind.Equal))
{
    string joined = string.Join("\n", hunk.Lines);
    Console.WriteLine($"  {hunk.Kind} @ old:{hunk.OldStart} new:{hunk.NewStart}");
}

Macro Recording (Script DSL)

Record user actions as a script and replay them on other documents:

using TextAPI.Core.Scripting;

// Script recorded from user actions (could come from a macro recorder UI)
const string MACRO = """
    REPLACE_ALL "System.Console.WriteLine" "Console.WriteLine"
    FIND "namespace " /i
    INSERT_AT 0 "// Auto-processed\n"
    GOTO 1 0
""";

// Apply to a list of files
async Task ApplyMacroAsync(string[] files)
{
    // Pre-parse once
    var cmds = ScriptParser.Parse(MACRO);

    foreach (var path in files)
    {
        var doc    = new TextDocument();
        await doc.LoadFileAsync(path);

        var result = new ScriptRunner(doc).Execute(cmds);

        if (result.Success)
            await doc.SaveFileAsync();
        else
            Console.WriteLine($"{path}: error at line {result.ErrorLine}: {result.ErrorMessage}");
    }
}

Interactive REPL Session

Wire up a simple read-eval-print loop:

using TextAPI.Repl;
using TextAPI.Core.Cursor;

var doc  = new TextDocument();
var mc   = new MultiCursor(doc);
var host = new CSharpScriptHost(doc, mc);

await doc.LoadFileAsync("file.cs");

Console.WriteLine("TextAPI REPL. Type C# code. Empty line to exit.");

var sb = new System.Text.StringBuilder();
while (true)
{
    string? line = Console.ReadLine();
    if (string.IsNullOrWhiteSpace(line)) break;

    sb.AppendLine(line);
    var r = await host.ExecuteAsync(sb.ToString());
    sb.Clear();

    if (!string.IsNullOrEmpty(r.Output))
        Console.Write(r.Output);

    if (!r.Success)
        Console.WriteLine($"Error: {r.ErrorMessage}");
}

Change-Tracking Gutter

Maintain a gutter display showing Added/Modified/Deleted markers as the user edits:

using TextAPI.Core.ChangeTracking;

var tracker = new ChangeTracker(doc);

// Capture baseline on load
await doc.LoadFileAsync("file.cs");
tracker.SetBaseline();

// Re-render gutter after every edit
tracker.ChangesUpdated += (_, _) =>
{
    for (int i = 0; i < doc.LineCount; i++)
    {
        char marker = tracker.GetStatus(i) switch
        {
            LineStatus.Added    => '+',
            LineStatus.Modified => '~',
            _                    => ' '
        };
        // Update gutter cell for line i with marker
    }
};

// Re-baseline on Ctrl+S
async Task SaveAsync()
{
    await doc.SaveFileAsync();
    tracker.SetBaseline();
}

Snippet Expansion (Tab Stop Navigation)

Full snippet lifecycle — insert, navigate tab stops, handle mirrors:

using TextAPI.Core.Snippets;
using TextAPI.Core.Cursor;

var cursor   = new TextCursor(doc);
var template = SnippetEngine.Parse("""
    for (int $1 = 0; $1 < $2; $1++) {
        $0
    }
""");

var session  = SnippetEngine.BeginSnippet(doc, template, cursor.CaretOffset);

// Tab key handler
bool OnTab()
{
    TabStop? stop = session.NextTabStop();
    if (stop is null)
    {
        session.Commit();  // reached $0 — done
        return false;
    }
    cursor.SetSelection(stop.Value.Offset, stop.Value.Offset + stop.Value.Length);
    return true;
}

// User types into a tab stop — update mirrors
void OnUserType(string text)
{
    cursor.InsertText(text);
    session.UpdateTabStop(session.CurrentTabStopIndex, cursor.SelectedText);
}

// Escape — cancel and undo
void OnEscape() => session.Cancel();
