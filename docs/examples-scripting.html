Script DSL & REPL — Examples

Every script command with all flag combinations, ScriptResult inspection, and every CSharpScriptHost / ScriptGlobals feature.

Script DSL

Runner setup

using TextAPI.Core.Scripting;

var doc    = new TextDocument();
doc.Load("Hello World\nLine 2\nLine 3");

var runner = new ScriptRunner(doc);

MOVE <offset>

var r = runner.Run("MOVE 6");
Console.WriteLine(runner.CursorOffset); // 6
Console.WriteLine(r.CursorOffset);        // 6

GOTO <line> [<col>]

// GOTO uses 1-based line numbers, 0-based column
runner.Run("GOTO 2");     // line 2 (1-based), col 0
Console.WriteLine(runner.CursorOffset); // 12  (start of "Line 2")

runner.Run("GOTO 2 5");   // line 2, col 5
Console.WriteLine(runner.CursorOffset); // 17

INSERT "<text>"

runner.Run("""
    GOTO 1 0
    INSERT "// Header\n"
""");
Console.WriteLine(doc.GetLine(0)); // "// Header"
// Insert with escape sequences
runner.Run("""MOVE 0
INSERT "line1\nline2\n"
""");

INSERT_AT <offset> "<text>"

doc.Load("World");
runner.Run("INSERT_AT 0 \"Hello, \"");
Console.WriteLine(doc.GetText()); // "Hello, World"

DELETE <n>

doc.Load("Hello World");
runner.Run("""
    MOVE 5
    DELETE 6
""");
Console.WriteLine(doc.GetText()); // "Hello"

DELETE_AT <offset> <n>

doc.Load("Hello, World");
runner.Run("DELETE_AT 5 2"); // delete ", " at offset 5
Console.WriteLine(doc.GetText()); // "HelloWorld"

DELETE_LINE [<line>]

doc.Load("a\nb\nc\nd");

// Delete line 2 (1-based)
runner.Run("DELETE_LINE 2");
Console.WriteLine(doc.GetText()); // "a\nc\nd"

// Delete cursor's current line (default when no arg)
runner.Run("""
    GOTO 2
    DELETE_LINE
""");
Console.WriteLine(doc.GetText()); // "a\nd"

SELECT <start> <end>

doc.Load("Hello World");
runner.Run("SELECT 6 11");
Console.WriteLine(runner.CursorOffset); // 11  (active end)
// Use in combination with INSERT to replace selection:
runner.Run("""
    SELECT 6 11
    INSERT "TextAPI"
""");
Console.WriteLine(doc.GetText()); // "Hello TextAPI"

REPLACE_ALL "<pat>" "<rep>" [flags]

doc.Load("foo Foo FOO");

// Case-sensitive (default)
runner.Run("REPLACE_ALL \"foo\" \"bar\"");
Console.WriteLine(doc.GetText()); // "bar Foo FOO"

// Case-insensitive /i
doc.Load("foo Foo FOO");
runner.Run("REPLACE_ALL \"foo\" \"bar\" /i");
Console.WriteLine(doc.GetText()); // "bar bar bar"

// Whole word /w
doc.Load("cat concatenate cats");
runner.Run("REPLACE_ALL \"cat\" \"dog\" /w");
Console.WriteLine(doc.GetText()); // "dog concatenate cats"

// Regex /r
doc.Load("2024-01-15");
runner.Run("REPLACE_ALL \"(\\d{4})-(\\d{2})-(\\d{2})\" \"$3/$2/$1\" /r");
Console.WriteLine(doc.GetText()); // "15/01/2024"

// Combined /i /w
runner.Run("REPLACE_ALL \"hello\" \"world\" /i /w");
// LastReplaceCount after REPLACE_ALL
var r = runner.Run("REPLACE_ALL \"foo\" \"bar\"");
Console.WriteLine(r.LastReplaceCount); // number of replacements

FIND "<pat>" [flags]

doc.Load("one two one three");
runner.Run("MOVE 0");

runner.Run("FIND \"one\"");
Console.WriteLine(runner.CursorOffset); // 0  (first match)

runner.Run("FIND \"one\"");
Console.WriteLine(runner.CursorOffset); // 8  (second match)

// Case-insensitive
doc.Load("Hello HELLO hello");
runner.Run("MOVE 0");
runner.Run("FIND \"hello\" /i");
Console.WriteLine(runner.CursorOffset); // 0

// Regex
doc.Load("abc 123 def 456");
runner.Run("MOVE 0");
runner.Run("FIND \"\\d+\" /r");
Console.WriteLine(runner.CursorOffset); // 4

FIND_PREV "<pat>" [flags]

doc.Load("one two one three");
runner.Run("MOVE 17"); // end

runner.Run("FIND_PREV \"one\"");
Console.WriteLine(runner.CursorOffset); // 8

runner.Run("FIND_PREV \"one\"");
Console.WriteLine(runner.CursorOffset); // 0

FIND_ALL "<pat>" [flags]

doc.Load("TODO fix\nTODO test\nDONE ship");
var r = runner.Run("FIND_ALL \"TODO\"");

Console.WriteLine(r.LastFindAll.Count); // 2
foreach (var m in r.LastFindAll)
    Console.WriteLine($"{m.Offset}+{m.Length}");
// 0+4
// 9+4

// Regex FIND_ALL
r = runner.Run("FIND_ALL \"\\bT\\w+\" /r /i");
Console.WriteLine(r.LastFindAll.Count); // "TODO" twice + "test"

UNDO and REDO

doc.Load("original");
runner.Run("INSERT_AT 0 \"NEW \"");
Console.WriteLine(doc.GetText()); // "NEW original"

runner.Run("UNDO");
Console.WriteLine(doc.GetText()); // "original"

runner.Run("REDO");
Console.WriteLine(doc.GetText()); // "NEW original"

NOP

// Useful as a conditional placeholder
runner.Run("""
    FIND "marker"
    NOP
""");
// NOP always succeeds, does nothing

ScriptResult — Every Property

Success and ErrorMessage

doc.Load("test");
var r = runner.Run("""
    MOVE 0
    DELETE 4
    MOVE 9999
""");

Console.WriteLine(r.Success);       // false
Console.WriteLine(r.ErrorMessage);  // "offset out of range"
Console.WriteLine(r.ErrorLine);    // 3  (script line number, 1-based)
Console.WriteLine(r.StepsExecuted); // 3  (all 3 lines ran, including the failing one)
Console.WriteLine(r.CursorOffset); // offset at point of failure

StepsExecuted and Steps

doc.Load("Hello World");
var r = runner.Run("""
    MOVE 0
    REPLACE_ALL "Hello" "Hi"
    FIND "World"
""");

Console.WriteLine(r.StepsExecuted); // 3
foreach (var step in r.Steps)
{
    Console.WriteLine($"L{step.LineNumber} {step.Verb}");
    Console.WriteLine($"  ok={step.Success} caret={step.CursorAfter}");
    if (step.ErrorMessage != null)
        Console.WriteLine($"  error: {step.ErrorMessage}");
}

LastReplaceCount and LastFindAll

doc.Load("a b a c a");
var r = runner.Run("""
    REPLACE_ALL "a" "X"
    FIND_ALL "X"
""");

Console.WriteLine(r.LastReplaceCount); // 3
Console.WriteLine(r.LastFindAll.Count); // 3
Console.WriteLine(r.LastFindAll[0].Offset); // 0

Pre-parsed execution — parse once, run many

var script = """
    REPLACE_ALL "TODO" "DONE"
    TRIM_TRAILING_WHITESPACE
""";

var commands = ScriptParser.Parse(script); // IReadOnlyList<ScriptCommand>

foreach (var path in files)
{
    var d = new TextDocument();
    await d.LoadFileAsync(path);
    var r2 = new ScriptRunner(d).Execute(commands); // no re-parse
    if (r2.Success) await d.SaveFileAsync();
}

C# REPL (CSharpScriptHost)

Setup

using TextAPI.Repl;
using TextAPI.Core.Cursor;

var doc  = new TextDocument();
var mc   = new MultiCursor(doc);
var host = new CSharpScriptHost(doc, mc);

ExecuteAsync — basic

doc.Load("Hello World\nLine 2");
var r = await host.ExecuteAsync("Print(doc.LineCount);");

Console.WriteLine(r.Success);  // true
Console.WriteLine(r.Output);   // "2"

ScriptExecutionResult — all properties

// Success run
var r = await host.ExecuteAsync("1 + 1");
Console.WriteLine(r.Success);       // true
Console.WriteLine(r.ReturnValue);   // 2
Console.WriteLine(r.Output);        // ""  (no Print)
Console.WriteLine(r.ErrorMessage);  // null
Console.WriteLine(r.Exception);     // null

// Failed run
var r2 = await host.ExecuteAsync("throw new Exception(\"oops\");");
Console.WriteLine(r2.Success);       // false
Console.WriteLine(r2.ErrorMessage);  // "oops"
Console.WriteLine(r2.Exception?.GetType().Name); // "Exception"

Print / print globals

var r = await host.ExecuteAsync("""
    Print("line 1");
    print("line 2");
    Print();         // blank line
    Print(42);       // any object
    Print(null);     // prints blank line
""");
Console.WriteLine(r.Output);
// line 1
// line 2
//
// 42
//

doc global — full document access

await host.ExecuteAsync("""
    doc.Load("Hello\nWorld\nFoo");
    Print(doc.LineCount);   // 3
    Print(doc.Length);      // 15
    Print(doc.GetLine(0));  // "Hello"
    doc.Insert(0, "// ");
    Print(doc.GetLine(0));  // "// Hello"
    doc.Undo();
    Print(doc.GetLine(0));  // "Hello"
""");

mc global — multi-cursor

await host.ExecuteAsync("""
    doc.Load("line1\nline2\nline3");
    mc.AddColumnSelection(0, 2, 0);  // cursor on each line at col 0
    mc.InsertText("// ");             // comment all 3 lines
    Print(doc.GetLine(0));           // "// line1"
""");

NewPipeline() global

var r = await host.ExecuteAsync("""
    doc.Load("TODO fix\nTODO test");
    var result = NewPipeline()
        .Add(new ReplaceAllOperation { Find = "TODO", Replace = "DONE" })
        .Add(new TrimTrailingWhitespaceOperation())
        .Execute();
    Print(result.Success);
    Print(result.AuditLog.Summary);
""");
Console.WriteLine(r.Output);
// True
// 2 op(s) succeeded; +8/-8 chars

RunJson() global

var r = await host.ExecuteAsync("""
    doc.Load("foo bar foo");
    var pr = RunJson(@"{
      ""operations"": [
        { ""type"": ""REPLACE_ALL"", ""find"": ""foo"", ""replace"": ""qux"" }
      ]
    }");
    Print(pr.Success);
    Print(doc.GetText());
""");
Console.WriteLine(r.Output);
// True
// qux bar qux

ParseOps() global

var r = await host.ExecuteAsync("""
    var ops = ParseOps(@"{
      ""operations"": [
        { ""type"": ""APPEND"", ""text"": ""\n// EOF"" }
      ]
    }");
    Print(ops.Count);                // 1
    Print(ops[0].GetType().Name);    // "AppendOperation"
    // Inspect without executing — useful for preview/validation
""");

Persistent state across submissions

// Submission 1 — declare helper
await host.ExecuteAsync("""
    int CountTodos() => doc.CountMatches("TODO");
""");

// Submission 2 — use helper (still in scope)
var r = await host.ExecuteAsync("""
    Print(CountTodos());
""");
Console.WriteLine(r.Output); // e.g. "3"

// Submission 3 — add another variable
await host.ExecuteAsync("""
    var prefix = "RESOLVED: ";
    doc.ReplaceAll("TODO", prefix + "TODO");
""");

// Submission 4 — prefix still in scope
await host.ExecuteAsync("Print(prefix);"); // "RESOLVED: "

HasState

Console.WriteLine(host.HasState); // false (fresh host)
await host.ExecuteAsync("var x = 1;");
Console.WriteLine(host.HasState); // true

Reset() — clear all session state

await host.ExecuteAsync("var secretVar = 42;");
Console.WriteLine(host.HasState); // true

host.Reset();
Console.WriteLine(host.HasState);  // false

// secretVar is gone
var r = await host.ExecuteAsync("Print(secretVar);");
Console.WriteLine(r.Success);       // false — CS0103 name not found

Cancellation

using var cts = new CancellationTokenSource(TimeSpan.FromSeconds(3));

var r = await host.ExecuteAsync("""
    while (true) { }  // infinite loop
""", cts.Token);

Console.WriteLine(r.Success);      // false
Console.WriteLine(r.ErrorMessage); // "The operation was canceled."

Pre-imported namespaces (no using needed)

// All of these work without explicit using directives:
await host.ExecuteAsync("""
    // System
    var list = new List<int> { 1, 2, 3 };
    var query = list.Where(x => x > 1).ToList();   // System.Linq

    // TextAPI
    var d = new TextDocument();
    var opts = new SearchOptions { UseRegex = true };
    var m = new MultiCursor(d);

    // Operations
    var op = new ReplaceAllOperation { Find = "a", Replace = "b" };
    var pl = new DocumentPipeline(d);

    Print(query.Count);   // 2
""");

Multi-line expression return value

// The last expression in a submission is the ReturnValue
var r = await host.ExecuteAsync("""
    var lines = doc.GetText().Split('\n');
    lines.Length
""");
Console.WriteLine(r.ReturnValue); // number of lines (int)
