All 27 Operations — Examples

A focused, runnable example for every operation — every property, its default, and edge-case behaviour.

Offset Operations

InsertAtOperation

// Insert "START " at the very beginning
var doc = new TextDocument(); doc.Load("World");

var result = new DocumentPipeline(doc)
    .Add(new InsertAtOperation(0, "Hello, "))
    .Execute();

Console.WriteLine(doc.GetText());           // "Hello, World"
Console.WriteLine(result.Success);          // true
Console.WriteLine(result.AuditLog.Summary); // "1 op(s) succeeded; +7/-0 chars"
// Failure: offset out of range
var r = new DocumentPipeline(doc)
    .Add(new InsertAtOperation(9999, "x"))
    .Execute();
Console.WriteLine(r.Success);       // false
Console.WriteLine(r.ErrorMessage);  // offset out of range
Console.WriteLine(r.WasRolledBack); // true

DeleteAtOperation

doc.Load("Hello, World!");

// Delete the comma and space (", " = 2 chars at offset 5)
new DocumentPipeline(doc)
    .Add(new DeleteAtOperation(5, 2))
    .Execute();
Console.WriteLine(doc.GetText()); // "HelloWorld!"
// Delete last character
new DocumentPipeline(doc)
    .Add(new DeleteAtOperation(doc.Length - 1, 1))
    .Execute();

ReplaceAtOperation

doc.Load("Hello World");

// Replace "World" (offset 6, length 5) with "TextAPI"
new DocumentPipeline(doc)
    .Add(new ReplaceAtOperation(6, 5, "TextAPI"))
    .Execute();
Console.WriteLine(doc.GetText()); // "Hello TextAPI"

Anchor Operations

InsertAfterOperation

doc.Load("namespace Foo { class Bar { } }");

// Insert after the opening of class Bar
new DocumentPipeline(doc)
    .Add(new InsertAfterOperation
    {
        Anchor        = "class Bar {",
        Text          = " int x = 1;",
        Occurrence    = 1,      // first (and only) occurrence
        CaseSensitive = true
    })
    .Execute();
Console.WriteLine(doc.GetText());
// "namespace Foo { class Bar { int x = 1; } }"
// 2nd occurrence of "end"
doc.Load("end first end second end third");
new DocumentPipeline(doc)
    .Add(new InsertAfterOperation { Anchor = "end", Text = "[2]", Occurrence = 2 })
    .Execute();
Console.WriteLine(doc.GetText()); // "end first end[2] second end third"

ReplaceSectionOperation

doc.Load("BEGIN old content END rest");

// Replace between anchors (not including them)
new DocumentPipeline(doc)
    .Add(new ReplaceSectionOperation
    {
        StartAnchor     = "BEGIN",
        EndAnchor       = "END",
        Text            = " NEW CONTENT ",
        IncludeAnchors  = false     // default: keep the anchors
    })
    .Execute();
Console.WriteLine(doc.GetText()); // "BEGIN NEW CONTENT END rest"
// Replace including anchors
doc.Load("[[ old ]] rest");
new DocumentPipeline(doc)
    .Add(new ReplaceSectionOperation
    {
        StartAnchor    = "[[", EndAnchor = "]]",
        Text           = "{{new}}",
        IncludeAnchors = true
    })
    .Execute();
Console.WriteLine(doc.GetText()); // "{{new}} rest"

Pattern Operations

ReplaceAllOperation

doc.Load("foo bar foo baz FOO");

// Case-sensitive (default)
var r = new DocumentPipeline(doc)
    .Add(new ReplaceAllOperation { Find = "foo", Replace = "qux" })
    .Execute();
Console.WriteLine(doc.GetText());                       // "qux bar qux baz FOO"
Console.WriteLine(r.OperationResults[0].MatchCount); // 2
// Case-insensitive
doc.Load("foo Foo FOO");
new DocumentPipeline(doc)
    .Add(new ReplaceAllOperation
    {
        Find          = "foo",
        Replace       = "bar",
        CaseSensitive = false
    })
    .Execute();
Console.WriteLine(doc.GetText()); // "bar bar bar"
// Whole word only
doc.Load("cat concatenate cats catfish");
new DocumentPipeline(doc)
    .Add(new ReplaceAllOperation
    {
        Find      = "cat",
        Replace   = "dog",
        WholeWord = true
    })
    .Execute();
Console.WriteLine(doc.GetText()); // "dog concatenate cats catfish"

RegexReplaceOperation

doc.Load("2024-01-15 and 2025-06-30");

// Reformat dates from YYYY-MM-DD to DD/MM/YYYY
new DocumentPipeline(doc)
    .Add(new RegexReplaceOperation
    {
        Pattern     = @"(\d{4})-(\d{2})-(\d{2})",
        Replacement = "$3/$2/$1"
    })
    .Execute();
Console.WriteLine(doc.GetText()); // "15/01/2024 and 30/06/2025"
// Remove trailing semicolons
doc.Load("int x = 1;\nstring s = \"hello\";\nvar y = 2");
new DocumentPipeline(doc)
    .Add(new RegexReplaceOperation { Pattern = @";$", Replacement = "" })
    .Execute();
Console.WriteLine(doc.GetText());
// "int x = 1
// string s = "hello"
// var y = 2"
// Invalid regex → failure (pipeline rolls back)
var r = new DocumentPipeline(doc)
    .Add(new RegexReplaceOperation { Pattern = "[invalid", Replacement = "" })
    .Execute();
Console.WriteLine(r.Success);      // false
Console.WriteLine(r.ErrorMessage); // invalid regex message

Line Operations

InsertLineOperation

doc.Load("line0\nline1\nline2");

// Insert before line 1 (0-based) — pushes line1 down
new DocumentPipeline(doc)
    .Add(new InsertLineOperation { LineIndex = 1, Text = "inserted" })
    .Execute();
Console.WriteLine(doc.GetText());
// line0
// inserted
// line1
// line2
// Insert at beginning (LineIndex = 0)
new DocumentPipeline(doc)
    .Add(new InsertLineOperation { LineIndex = 0, Text = "// First line" })
    .Execute();
// Insert at end (LineIndex = doc.LineCount)
new DocumentPipeline(doc)
    .Add(new InsertLineOperation { LineIndex = doc.LineCount, Text = "// EOF" })
    .Execute();

DeleteLineOperation

doc.Load("a\nb\nc\nd");

// Delete line index 1 ("b")
new DocumentPipeline(doc)
    .Add(new DeleteLineOperation { LineIndex = 1 })
    .Execute();
Console.WriteLine(doc.GetText()); // "a\nc\nd"
// Delete last line
new DocumentPipeline(doc)
    .Add(new DeleteLineOperation { LineIndex = doc.LineCount - 1 })
    .Execute();

ReplaceLineOperation

doc.Load("foo\nbar\nbaz");

// Replace line 1 ("bar") with new content
new DocumentPipeline(doc)
    .Add(new ReplaceLineOperation { LineIndex = 1, Text = "REPLACED" })
    .Execute();
Console.WriteLine(doc.GetText()); // "foo\nREPLACED\nbaz"

InsertAfterLineMatchOperation

doc.Load("using System;\nusing TextAPI.Core;\n\nvar doc = new TextDocument();");

// Insert a new using after the line containing "TextAPI.Core"
new DocumentPipeline(doc)
    .Add(new InsertAfterLineMatchOperation
    {
        Pattern  = "TextAPI.Core",
        LineText = "using TextAPI.Operations;"
    })
    .Execute();
Console.WriteLine(doc.GetLine(2)); // "using TextAPI.Operations;"
// Regex match — insert after 2nd blank line
new DocumentPipeline(doc)
    .Add(new InsertAfterLineMatchOperation
    {
        Pattern    = @"^\s*$",  // blank line
        LineText   = "// inserted after blank",
        UseRegex   = true,
        Occurrence = 2
    })
    .Execute();

ConvertCaseOperation — Upper / Lower / TitleCase

doc.Load("hello world");

// Uppercase entire document
new DocumentPipeline(doc)
    .Add(new ConvertCaseOperation { Mode = CaseMode.Upper })
    .Execute();
Console.WriteLine(doc.GetText()); // "HELLO WORLD"
// TitleCase a specific range between anchors
doc.Load("prefix [title goes here] suffix");
new DocumentPipeline(doc)
    .Add(new ConvertCaseOperation
    {
        Mode        = CaseMode.TitleCase,
        StartAnchor = "[",
        EndAnchor   = "]"
    })
    .Execute();
Console.WriteLine(doc.GetText()); // "prefix [Title Goes Here] suffix"

SortLinesOperation

doc.Load("banana\napple\ncherry\ndate");

// Sort all lines ascending
new DocumentPipeline(doc)
    .Add(new SortLinesOperation())
    .Execute();
Console.WriteLine(doc.GetText());
// apple
// banana
// cherry
// date
// Sort descending, case-insensitive, lines 1–3 only
doc.Load("HEADER\nbanana\napple\ncherry\nFOOTER");
new DocumentPipeline(doc)
    .Add(new SortLinesOperation
    {
        StartLine     = 1,
        EndLine       = 3,
        Descending    = true,
        CaseSensitive = false
    })
    .Execute();
Console.WriteLine(doc.GetText());
// HEADER
// cherry
// banana
// apple
// FOOTER

DeduplicateLinesOperation

doc.Load("apple\nbanana\napple\ncherry\nbanana\napple");

// Remove all duplicates (keep first occurrence)
new DocumentPipeline(doc)
    .Add(new DeduplicateLinesOperation())
    .Execute();
Console.WriteLine(doc.GetText());
// apple
// banana
// cherry
// ConsecutiveOnly — only remove adjacent duplicates (like Unix uniq)
doc.Load("a\na\nb\na\nb\nb");
new DocumentPipeline(doc)
    .Add(new DeduplicateLinesOperation { ConsecutiveOnly = true })
    .Execute();
Console.WriteLine(doc.GetText());
// a
// b
// a
// b

IndentOperation

doc.Load("line0\nline1\nline2\nline3");

// Indent lines 1–2 with 4 spaces
new DocumentPipeline(doc)
    .Add(new IndentOperation

    .Execute();

Query Operations

FindAllOperation (non-mutating)

doc.Load("TODO fix login\nTODO add tests\ndone deploy");

var r = new DocumentPipeline(doc)
    .Add(new FindAllOperation { Pattern = "TODO" })
    .Execute();

Console.WriteLine(r.OperationResults[0].MatchCount); // 2
Console.WriteLine(doc.GetText());                    // unchanged!
// Regex variant
new DocumentPipeline(doc)
    .Add(new FindAllOperation
    {
        Pattern       = @"\bTODO\b",
        UseRegex      = true,
        CaseSensitive = false
    })
    .Execute();

ContainsOperation

doc.Load("production deployment");

var r = new DocumentPipeline(doc)
    .Add(new ContainsOperation { Pattern = "prod" })
    .Execute();

Console.WriteLine(r.OperationResults[0].MatchCount); // 1 (found)
// 0 means not found — but Success is still true (Contains never fails the pipeline)
// Use it as a guard — fail explicitly if not found
var opResult = r.OperationResults[0];
if (opResult.MatchCount == 0)
    Console.WriteLine("Pattern not present — skipping pipeline");

ExtractSectionOperation (non-mutating)

doc.Load("HEADER\n/* config */\nkey=value\n/* /config */\nFOOTER");

var r = new DocumentPipeline(doc)
    .Add(new ExtractSectionOperation
    {
        StartAnchor    = "/* config */",
        EndAnchor      = "/* /config */",
        IncludeAnchors = false
    })
    .Execute();

string extracted = r.OperationResults[0].ExtractedText!;
Console.WriteLine(extracted); // "\nkey=value\n"
Console.WriteLine(doc.GetText()); // unchanged!

Structural Operations

MoveSectionOperation

doc.Load("AAA\n-- SECTION --\ncontent\n-- END --\nBBB\n-- TARGET --\nCCC");

new DocumentPipeline(doc)
    .Add(new MoveSectionOperation
    {
        SourceStartAnchor      = "-- SECTION --",
        SourceEndAnchor        = "-- END --",
        DestinationAnchor      = "-- TARGET --",
        InsertAfterDestination = true   // default
    })
    .Execute();
// section (minus anchors) is cut from original position and inserted after "-- TARGET --"

SwapSectionsOperation

doc.Load("[A]Hello[/A] ... [B]World[/B]");

new DocumentPipeline(doc)
    .Add(new SwapSectionsOperation
    {
        Section1Start = "[A]", Section1End = "[/A]",
        Section2Start = "[B]", Section2End = "[/B]"
    })
    .Execute();
Console.WriteLine(doc.GetText()); // "[A]World[/A] ... [B]Hello[/B]"
// Failure: sections overlap
doc.Load("[A] [B] [/A] [/B]");
var r = new DocumentPipeline(doc)
    .Add(new SwapSectionsOperation
    {
        Section1Start = "[A]", Section1End = "[/A]",
        Section2Start = "[B]", Section2End = "[/B]"
    })
    .Execute();
Console.WriteLine(r.Success);      // false
Console.WriteLine(r.WasRolledBack); // true

EnsureTrailingNewlineOperation

doc.Load("last line with no newline");
new DocumentPipeline(doc)
    .Add(new EnsureTrailingNewlineOperation())
    .Execute();
Console.WriteLine(doc.GetText().EndsWith("\n")); // true

// If already ends with \n, no-op
doc.Load("already ends\n");
new DocumentPipeline(doc).Add(new EnsureTrailingNewlineOperation()).Execute();
Console.WriteLine(doc.GetText() == "already ends\n"); // true (unchanged)
