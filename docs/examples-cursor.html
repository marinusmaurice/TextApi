Cursors — Examples

Every TextCursor and MultiCursor method demonstrated, including all movement, selection, editing, column selection, and paste variants.

TextCursor

Construction

using TextAPI.Core.Cursor;

var doc    = new TextDocument();
doc.Load("Hello\nWorld\nFoo");

var cursor = new TextCursor(doc);            // starts at offset 0
var cursor2 = new TextCursor(doc, offset: 6); // starts at "W"
Console.WriteLine(cursor2.CaretLine);   // 1
Console.WriteLine(cursor2.CaretColumn); // 0

Position properties

doc.Load("abc\nde\nfg");
var cursor = new TextCursor(doc, offset: 5); // "e" on line 1

Console.WriteLine(cursor.CaretOffset);   // 5
Console.WriteLine(cursor.ActiveOffset);  // 5  (alias)
Console.WriteLine(cursor.AnchorOffset);  // 5  (no selection)
Console.WriteLine(cursor.CaretLine);     // 1
Console.WriteLine(cursor.CaretColumn);   // 1
Console.WriteLine(cursor.HasSelection);  // false

MoveTo — collapse to absolute offset

cursor.MoveTo(0);
Console.WriteLine(cursor.CaretOffset);   // 0
Console.WriteLine(cursor.HasSelection);  // false

SelectTo — extend selection

cursor.MoveTo(0);
cursor.SelectTo(5);
Console.WriteLine(cursor.AnchorOffset);  // 0
Console.WriteLine(cursor.ActiveOffset);  // 5
Console.WriteLine(cursor.SelectionStart); // 0
Console.WriteLine(cursor.SelectionEnd);   // 5
Console.WriteLine(cursor.SelectedText);   // "abc\nd"

SetSelection — anchor and active explicitly

// Backward selection (active < anchor)
cursor.SetSelection(anchor: 5, active: 0);
Console.WriteLine(cursor.SelectionStart); // 0
Console.WriteLine(cursor.SelectionEnd);   // 5
Console.WriteLine(cursor.ActiveOffset);  // 0  (caret at start of selection)

CollapseToStart / CollapseToEnd

cursor.SetSelection(0, 5);

cursor.CollapseToStart();
Console.WriteLine(cursor.CaretOffset);  // 0
Console.WriteLine(cursor.HasSelection); // false

cursor.SetSelection(0, 5);
cursor.CollapseToEnd();
Console.WriteLine(cursor.CaretOffset);  // 5

MoveLeft / MoveRight

doc.Load("Hello");
cursor.MoveTo(3);

cursor.MoveLeft();   // one grapheme left
Console.WriteLine(cursor.CaretOffset); // 2

cursor.MoveRight(2); // two graphemes right
Console.WriteLine(cursor.CaretOffset); // 4

MoveWordLeft / MoveWordRight

doc.Load("hello world foo");
cursor.MoveTo(15); // end

cursor.MoveWordLeft();
Console.WriteLine(cursor.CaretOffset); // 12  (start of "foo")

cursor.MoveWordLeft();
Console.WriteLine(cursor.CaretOffset); // 6   (start of "world")

cursor.MoveWordRight();
Console.WriteLine(cursor.CaretOffset); // 11  (after "world")

MoveToLineStart / MoveToLineEnd

doc.Load("    indented line");
cursor.MoveTo(10);

cursor.MoveToLineStart();
Console.WriteLine(cursor.CaretColumn); // 0

cursor.MoveToLineEnd();
Console.WriteLine(cursor.CaretOffset);  // doc.Length

MoveToDocumentStart / MoveToDocumentEnd

cursor.MoveToDocumentEnd();
Console.WriteLine(cursor.CaretOffset); // doc.Length

cursor.MoveToDocumentStart();
Console.WriteLine(cursor.CaretOffset); // 0

SelectLeft / SelectRight

doc.Load("Hello");
cursor.MoveTo(3);
cursor.SelectLeft(2);
Console.WriteLine(cursor.SelectedText); // "el"  (offset 1–3)

cursor.MoveTo(0);
cursor.SelectRight(5);
Console.WriteLine(cursor.SelectedText); // "Hello"

SelectWordLeft / SelectWordRight

doc.Load("one two three");
cursor.MoveTo(13); // end of "three"

cursor.SelectWordLeft();
Console.WriteLine(cursor.SelectedText); // "three"

SelectToLineStart / SelectToLineEnd

doc.Load("Hello World");
cursor.MoveTo(5);
cursor.SelectToLineEnd();
Console.WriteLine(cursor.SelectedText); // " World"

cursor.MoveTo(5);
cursor.SelectToLineStart();
Console.WriteLine(cursor.SelectedText); // "Hello"

SelectToDocumentStart / SelectToDocumentEnd

cursor.MoveTo(5);
cursor.SelectToDocumentEnd();
Console.WriteLine(cursor.SelectedText); // everything from offset 5 onward

cursor.MoveTo(5);
cursor.SelectToDocumentStart();
Console.WriteLine(cursor.SelectedText); // everything from 0 to 5

SelectAll

cursor.SelectAll();
Console.WriteLine(cursor.SelectedText == doc.GetText()); // true

SelectLine

doc.Load("line0\nline1\nline2");
cursor.MoveTo(0);
cursor.SelectLine();               // selects caret's line (0)
Console.WriteLine(cursor.SelectedText); // "line0"

cursor.SelectLine(lineIndex: 2);    // select specific line by index
Console.WriteLine(cursor.SelectedText); // "line2"

SelectWordAtCaret

doc.Load("hello world");
cursor.MoveTo(7); // inside "world"
cursor.SelectWordAtCaret();
Console.WriteLine(cursor.SelectedText); // "world"

MoveUp / MoveDown

doc.Load("abc\nde\nfghij");
cursor.MoveTo(doc.PositionToOffset(2, 4)); // line 2, col 4 ("i")

cursor.MoveUp();
Console.WriteLine($"line={cursor.CaretLine} col={cursor.CaretColumn}"); // line=1 col=1 (clamped to "e")

cursor.MoveDown();
Console.WriteLine($"line={cursor.CaretLine} col={cursor.CaretColumn}"); // line=2 col=4 (preferred col restored)

SelectUp / SelectDown

doc.Load("aaaa\nbbbb\ncccc");
cursor.MoveTo(doc.PositionToOffset(1, 2)); // line 1, col 2
cursor.SelectDown();
Console.WriteLine(cursor.SelectedText); // "bb\ncc"  (from line 1 col 2 to line 2 col 2)

InsertText — replaces selection if any

doc.Load("Hello World");
cursor.SetSelection(6, 11); // "World"
cursor.InsertText("TextAPI");
Console.WriteLine(doc.GetText()); // "Hello TextAPI"

DeleteSelection

cursor.SetSelection(0, 6); // "Hello "
cursor.DeleteSelection();
Console.WriteLine(doc.GetText()); // "TextAPI"

DeleteLeft / DeleteRight (Backspace / Delete key)

doc.Load("abcde");
cursor.MoveTo(3);

cursor.DeleteLeft();    // backspace one char
Console.WriteLine(doc.GetText()); // "abde"

cursor.DeleteRight();    // delete one char
Console.WriteLine(doc.GetText()); // "abe"

cursor.DeleteLeft(2);    // backspace 2 chars
Console.WriteLine(doc.GetText()); // "e"

DeleteWordLeft / DeleteWordRight (Ctrl+Backspace / Ctrl+Delete)

doc.Load("hello world");
cursor.MoveTo(11); // end

cursor.DeleteWordLeft();
Console.WriteLine(doc.GetText()); // "hello "

cursor.MoveToDocumentEnd();  // still at end
cursor.DeleteWordLeft();
Console.WriteLine(doc.GetText()); // ""  — wait, "hello " has trailing space
// DeleteWordLeft removes the word group to the left including trailing non-word chars

Word boundary helpers

// Static IsWordChar
Console.WriteLine(TextCursor.IsWordChar('a'));  // true
Console.WriteLine(TextCursor.IsWordChar('_'));  // true
Console.WriteLine(TextCursor.IsWordChar('.'));  // false

// Instance helpers
doc.Load("hello world");
cursor.MoveTo(7);  // inside "world"
int wl = cursor.WordLeft(cursor.CaretOffset);
int wr = cursor.WordRight(cursor.CaretOffset);
Console.WriteLine($"word: [{wl},{wr}) = \"{doc.GetText(wl, wr - wl)}\"");
// word: [6,11) = "world"

MultiCursor

Construction and Count

var mc = new MultiCursor(doc);
Console.WriteLine(mc.Count);   // 1  (one cursor at offset 0)
Console.WriteLine(mc.Primary.CaretOffset); // 0

AddCursor — add at offset

doc.Load("line0\nline1\nline2");
var mc = new MultiCursor(doc);

mc.AddCursor(6);   // cursor at start of "line1"
mc.AddCursor(12);  // cursor at start of "line2"
Console.WriteLine(mc.Count);   // 3
Console.WriteLine(mc.Primary.CaretOffset); // 12  (last added is primary)

AddCursor — with selection

mc.AddCursor(anchor: 0, active: 5); // add cursor with "line0" selected
Console.WriteLine(mc.Primary.SelectedText); // "line0"

RemoveCursor

mc.RemoveCursor(0); // remove cursor at index 0 (sorted by offset)
Console.WriteLine(mc.Count); // 2

// No-op when only one cursor remains
while (mc.Count > 1) mc.RemoveCursor(0);
mc.RemoveCursor(0); // no-op
Console.WriteLine(mc.Count); // 1

Clear and SetSingle

mc.Clear();
Console.WriteLine(mc.Count);                   // 1, at offset 0
Console.WriteLine(mc.Primary.CaretOffset); // 0

mc.SetSingle(42);
Console.WriteLine(mc.Primary.CaretOffset); // 42

All — iterate cursors in order

doc.Load("a\nb\nc");
var mc = new MultiCursor(doc);
mc.AddCursor(4);
mc.AddCursor(2);

foreach (var c in mc.All)
    Console.WriteLine(c.CaretOffset);
// 0, 2, 4  — always ascending

AddColumnSelection

doc.Load("    alpha\n    beta\n    gamma\n    delta");
var mc = new MultiCursor(doc);

// One cursor per line (0–3) at column 0 — for block indent
mc.AddColumnSelection(startLine: 0, endLine: 3, column: 0);
Console.WriteLine(mc.Count); // 4

mc.InsertText("    "); // add 4 spaces at each cursor — 4 lines indented
Console.WriteLine(doc.GetLine(0)); // "        alpha"

doc.Undo(); // removes all 4 insertions in one step

InsertText — all cursors, one undo step

doc.Load("line1\nline2\nline3");
var mc = new MultiCursor(doc);
mc.AddCursor(0);
mc.AddCursor(6);
mc.AddCursor(12);

mc.InsertText("// ");
Console.WriteLine(doc.GetText());
// // line1
// // line2
// // line3

doc.Undo(); // single undo removes all three insertions

DeleteLeft / DeleteRight across all cursors

doc.Load("// line1\n// line2\n// line3");
var mc = new MultiCursor(doc);

// Place cursors after the "//" on each line
mc.SetSingle(2);
mc.AddCursor(11);
mc.AddCursor(20);

mc.DeleteLeft(3); // delete "// " at each cursor position
Console.WriteLine(doc.GetText());
// line1
// line2
// line3

DeleteSelection — all cursors

doc.Load("[DELETE]keep[DELETE]keep[DELETE]keep");
var mc = new MultiCursor(doc);
mc.SetSingle(0);
mc.Primary.SelectTo(8);  // select first "[DELETE]"
mc.AddCursor(anchor: 12, active: 20);
mc.AddCursor(anchor: 24, active: 32);
mc.DeleteSelection();
Console.WriteLine(doc.GetText()); // "keepkeepkeep"

SelectWordAtCaret — all cursors

doc.Load("foo bar foo bar");
var mc = new MultiCursor(doc);
mc.SetSingle(0);  // inside "foo"
mc.AddCursor(4);  // inside "bar"

mc.SelectWordAtCaret();
Console.WriteLine(mc.All[0].SelectedText); // "foo"
Console.WriteLine(mc.All[1].SelectedText); // "bar"

mc.InsertText("qux");  // replace both words simultaneously
Console.WriteLine(doc.GetText()); // "qux qux foo bar"

Paste — distributed vs broadcast

doc.Load("_\n_\n_");
var mc = new MultiCursor(doc);
mc.SetSingle(0);
mc.AddCursor(2);
mc.AddCursor(4);

// 3 cursors, 3 clipboard lines → distributed (each cursor gets one line)
mc.Paste(new[] { "alpha", "beta", "gamma" });
Console.WriteLine(doc.GetText());
// alpha_
// beta_
// gamma_
// Cursor count ≠ line count → broadcast (all cursors get same text)
doc.Load("_\n_\n_");
mc.SetSingle(0);
mc.AddCursor(2);
mc.AddCursor(4);
mc.Paste(new[] { "TEXT" });  // 1 line, 3 cursors → broadcast
Console.WriteLine(doc.GetText());
// TEXT_
// TEXT_
// TEXT_

Overlapping cursor merge

// After edits, cursors that overlap are automatically merged.
// This prevents double-applying edits to the same region.
doc.Load("abc");
var mc = new MultiCursor(doc);
mc.AddCursor(0);
mc.AddCursor(1);
mc.MoveRight(2); // both cursors move to offset 2 and 3
// they were adjacent/overlapping → merged to one cursor
Console.WriteLine(mc.Count); // 1
