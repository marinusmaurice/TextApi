Pipeline & Serialization — Examples

Every pipeline feature: building, executing, reading results, audit logs, offset drift, cancellation, rollback, and all JSON formats.

Building a Pipeline

Fluent chaining

using TextAPI.Operations;
using TextAPI.Operations.Pipeline;

var doc = new TextDocument();
doc.Load("TODO fix\nTODO test\ndone deploy");

PipelineResult result = new DocumentPipeline(doc)
    .Add(new ReplaceAllOperation  { Find = "TODO", Replace = "DONE" })
    .Add(new TrimTrailingWhitespaceOperation())
    .Add(new SortLinesOperation())
    .Execute();

Console.WriteLine(result.Success);           // true
Console.WriteLine(doc.GetText());
// DONE deploy
// DONE fix
// DONE test

Add IEnumerable of operations

var ops = new List<IDocumentOperation>
{
    new PrependOperation            { Text = "// generated\n" },
    new TrimTrailingWhitespaceOperation(),
    new EnsureTrailingNewlineOperation()
};

var pipeline = new DocumentPipeline(doc);
pipeline.Add(ops);  // add IEnumerable in one call
pipeline.Execute();

Re-run the same pipeline on multiple documents

var ops = new IDocumentOperation[]
{
    new TrimTrailingWhitespaceOperation(),
    new EnsureTrailingNewlineOperation()
};

foreach (var path in files)
{
    var doc = new TextDocument();
    await doc.LoadFileAsync(path);
    new DocumentPipeline(doc).Add(ops).Execute();
    if (doc.IsModified) await doc.SaveFileAsync();
}

PipelineResult — Every Property

Success, TextBefore, TextAfter

doc.Load("before");
var r = new DocumentPipeline(doc)
    .Add(new AppendOperation { Text = " after" })
    .Execute();

Console.WriteLine(r.Success);    // true
Console.WriteLine(r.TextBefore); // "before"
Console.WriteLine(r.TextAfter);  // "before after"

FailedOperationIndex and ErrorMessage

doc.Load("text");
var r = new DocumentPipeline(doc)
    .Add(new TrimTrailingWhitespaceOperation())         // op 0 — succeeds
    .Add(new InsertAtOperation(9999, "x"))               // op 1 — fails
    .Add(new EnsureTrailingNewlineOperation())          // op 2 — never runs
    .Execute();

Console.WriteLine(r.Success);                // false
Console.WriteLine(r.FailedOperationIndex);   // 1
Console.WriteLine(r.ErrorMessage);           // "offset out of range"
Console.WriteLine(r.WasRolledBack);          // true
Console.WriteLine(doc.GetText());            // "text" — pre-pipeline state

OperationResults — per-op results

doc.Load("TODO one\nTODO two\nDONE three");
var r = new DocumentPipeline(doc)
    .Add(new ReplaceAllOperation { Find = "TODO", Replace = "DONE" })
    .Add(new TrimTrailingWhitespaceOperation())
    .Execute();

foreach (var op in r.OperationResults)
{
    Console.WriteLine($"[{op.OperationType}]");
    Console.WriteLine($"  success={op.Success}");
    Console.WriteLine($"  +{op.CharsInserted}/-{op.CharsDeleted}");
    Console.WriteLine($"  matches={op.MatchCount}");
}

AuditLog — Every Property

Summary, AllSucceeded, totals

doc.Load("foo foo foo");
var r = new DocumentPipeline(doc)
    .Add(new ReplaceAllOperation { Find = "foo", Replace = "qux" })
    .Execute();

AuditLog log = r.AuditLog;
Console.WriteLine(log.Summary);             // "1 op(s) succeeded; +9/-9 chars"
Console.WriteLine(log.AllSucceeded);        // true
Console.WriteLine(log.TotalCharsInserted);  // 9  (3 × "qux")
Console.WriteLine(log.TotalCharsDeleted);   // 9  (3 × "foo")

AuditEntry — per-operation entry

foreach (AuditEntry entry in log.Entries)
{
    Console.WriteLine($"[{entry.TimestampUtc:HH:mm:ss.fff}] {entry.OperationType}");
    Console.WriteLine($"  success={entry.Success}");
    Console.WriteLine($"  +{entry.CharsInserted}/-{entry.CharsDeleted}");
    if (entry.MatchCount > 0)
        Console.WriteLine($"  matches={entry.MatchCount}");
    if (entry.ErrorMessage != null)
        Console.WriteLine($"  error: {entry.ErrorMessage}");
}

Using AuditLog for reporting

if (!r.AuditLog.AllSucceeded)
{
    var failed = r.AuditLog.Entries.First(e => !e.Success);
    Console.WriteLine($"First failure: {failed.OperationType} — {failed.ErrorMessage}");
}

int totalChanged = r.AuditLog.TotalCharsInserted + r.AuditLog.TotalCharsDeleted;
Console.WriteLine($"Total chars changed: {totalChanged}");

Rollback

Automatic rollback on failure

doc.Load("original text");
string snapshot = doc.GetText(); // capture before pipeline

var r = new DocumentPipeline(doc)
    .Add(new ReplaceAllOperation { Find = "original", Replace = "modified" })
    .Add(new InsertAtOperation(99999, "x"))  // will fail → rollback
    .Execute();

Console.WriteLine(r.WasRolledBack);  // true
Console.WriteLine(doc.GetText());    // "original text" — restored
Console.WriteLine(doc.GetText() == snapshot);  // true

WasRolledBack and TextBefore on failure

Console.WriteLine(r.WasRolledBack); // true
Console.WriteLine(r.TextBefore);  // pre-pipeline text (same as snapshot)
Console.WriteLine(r.TextAfter);   // equals TextBefore after rollback

Rollback calls doc.Load(snapshot) internally, which resets the undo stack. You cannot undo the failed pipeline's partial changes after a rollback.

Offset Drift Correction

Two InsertAt operations — drift auto-corrected

doc.Load("Hello World");
// Insert "FOO " at offset 0 (+4 chars), then "BAR " at original offset 6
// Without drift correction, "BAR " would land at "W" of World (offset 6).
// With drift correction, offset 6 is adjusted to 10 (6 + 4 drift).

new DocumentPipeline(doc)
    .Add(new InsertAtOperation(0,  "FOO "))   // +4 drift
    .Add(new InsertAtOperation(6,  "BAR "))   // auto-adjusted to offset 10
    .Execute();

Console.WriteLine(doc.GetText()); // "FOO Hello BAR World"

DeleteAt followed by InsertAt — negative drift

doc.Load("##Hello World");
new DocumentPipeline(doc)
    .Add(new DeleteAtOperation(0, 2))      // delete "##" → -2 drift
    .Add(new InsertAtOperation(7, "!"))    // original offset 7 → adjusted to 5
    .Execute();

Console.WriteLine(doc.GetText()); // "Hello! World"

Drift correction only applies to operations implementing IOffsetAwareOperation — the three offset-based ops (InsertAt, DeleteAt, ReplaceAt). Anchor-based operations re-resolve their anchors at execution time.

Cancellation

Cancel via CancellationToken

using var cts = new CancellationTokenSource();
cts.CancelAfter(TimeSpan.FromSeconds(5));

var r = pipeline.Execute(cts.Token);

if (!r.Success)
{
    Console.WriteLine(r.ErrorMessage);  // "Pipeline cancelled"
    Console.WriteLine(r.WasRolledBack); // true — document restored
}

Pre-cancelled token

var cts = new CancellationTokenSource();
cts.Cancel();  // cancel immediately

var r = pipeline.Execute(cts.Token);
Console.WriteLine(r.FailedOperationIndex); // -1 (no op ran)
Console.WriteLine(r.WasRolledBack);        // true

JSON Serialization

FromPipelineJson — pipeline wrapper format

using TextAPI.Operations.Serialization;

string json = """
{
  "operations": [
    { "type": "REPLACE_ALL", "find": "TODO", "replace": "DONE" },
    { "type": "TRIM_TRAILING_WHITESPACE" },
    { "type": "SORT_LINES", "descending": true }
  ]
}
""";

var ops    = OperationMapper.FromPipelineJson(json);
var result = new DocumentPipeline(doc).Add(ops).Execute();

FromOperationJson — single operation

string opJson = """{"type":"REPLACE_ALL","find":"foo","replace":"bar"}""";
IDocumentOperation op = OperationMapper.FromOperationJson(opJson);
new DocumentPipeline(doc).Add(op).Execute();

ToJson — serialize operation to JSON

var op   = new ReplaceAllOperation { Find = "old", Replace = "new", CaseSensitive = false };
string j = OperationMapper.ToJson(op);
Console.WriteLine(j);
// {"type":"REPLACE_ALL","find":"old","replace":"new","caseSensitive":false}

ToPipelineJson — serialize multiple operations

var ops = new IDocumentOperation[]
{
    new PrependOperation            { Text = "// header\n" },
    new TrimTrailingWhitespaceOperation()
};
string json = OperationMapper.ToPipelineJson(ops);
Console.WriteLine(json);
// {
//   "operations": [
//     {"type":"PREPEND","text":"// header\n"},
//     {"type":"TRIM_TRAILING_WHITESPACE"}
//   ]
// }

Deserialize unknown type — graceful failure

try
{
    OperationMapper.FromOperationJson("""{"type":"UNKNOWN_OP"}""");
}
catch (JsonException ex)
{
    Console.WriteLine(ex.Message); // "Unknown operation type: UNKNOWN_OP"
}

LLM Integration Pattern

// System prompt excerpt for an LLM tool definition:
// "Respond with JSON of the form { \"operations\": [...] }
//  using operation types: REPLACE_ALL, INSERT_AT, DELETE_AT, INSERT_LINE, ..."
//
// LLM response:
string llmResponse = """
{
  "operations": [
    { "type": "REPLACE_ALL", "find": "var ", "replace": "const " },
    { "type": "REGEX_REPLACE", "pattern": "console\\.log\\((.*?)\\)", "replacement": "logger.info($1)" },
    { "type": "ENSURE_TRAILING_NEWLINE" }
  ]
}
""";

try
{
    var ops    = OperationMapper.FromPipelineJson(llmResponse);
    var result = new DocumentPipeline(doc).Add(ops).Execute();

    if (result.Success)
        Console.WriteLine($"Applied: {result.AuditLog.Summary}");
    else
        // return error to LLM for retry
        Console.WriteLine($"Op #{result.FailedOperationIndex} failed: {result.ErrorMessage}");
}
catch (JsonException ex)
{
    Console.WriteLine($"LLM produced invalid JSON: {ex.Message}");
}
