Search, Decorations & Diff — Examples

Every SearchOptions combination, all search methods, every DecorationType, and all TextDiff / DiffChars variants.

SearchOptions — Every Combination

Default (case-sensitive, literal)

using TextAPI.Core.Search;

var doc = new TextDocument();
doc.Load("Hello hello HELLO");

var all = doc.FindAll("hello").ToList();
Console.WriteLine(all.Count); // 1  (only "hello", not "Hello" or "HELLO")

CaseSensitive = false

var opts = new SearchOptions { CaseSensitive = false };
var all  = doc.FindAll("hello", opts).ToList();
Console.WriteLine(all.Count); // 3

WholeWord = true

doc.Load("cat concatenate cats catfish");
var opts = new SearchOptions { WholeWord = true };
var all  = doc.FindAll("cat", opts).ToList();
Console.WriteLine(all.Count); // 1  (only the standalone "cat")
Console.WriteLine(doc.GetText(all[0].Offset, all[0].Length)); // "cat"

UseRegex = true

doc.Load("x = 1; y = 22; z = 333;");
var opts = new SearchOptions { UseRegex = true };
var nums = doc.FindAll(@"\d+", opts).ToList();

foreach (var m in nums)
    Console.WriteLine(doc.GetText(m.Offset, m.Length));
// 1
// 22
// 333

MaxResults — cap result count

doc.Load("a a a a a a a a a a");
var opts = new SearchOptions { MaxResults = 3 };
var all  = doc.FindAll("a", opts).ToList();
Console.WriteLine(all.Count); // 3  (stops after 3 even though 10 match)

CaseSensitive + WholeWord combined

doc.Load("Foo foo FOO foobar");
var opts = new SearchOptions { CaseSensitive = false, WholeWord = true };
var all  = doc.FindAll("foo", opts).ToList();
Console.WriteLine(all.Count); // 3  (Foo, foo, FOO — not foobar)

Search Methods

FindAll — lazy LINQ composition

doc.Load("TODO task1\nDONE task2\nTODO task3\nDONE task4");

// Take only first 2 matches
var first2 = doc.FindAll("TODO").Take(2).ToList();

// Only matches whose text at offset is uppercase T (trivial example)
var filtered = doc.FindAll("TODO").Where(m => m.Offset < 20).ToList();

// Convert matches to (line, col) for display
var positions = doc.FindAll("TODO")
    .Select(m => doc.OffsetToPosition(m.Offset))
    .ToList();
foreach (var (line, col) in positions)
    Console.WriteLine($"L{line + 1}:{col}");
// L1:0
// L3:0

FindFirst

doc.Load("one two one");
SearchMatch? first = doc.FindFirst("one");

Console.WriteLine(first?.Offset);    // 0
Console.WriteLine(first?.Length);    // 3

// Not found → null
SearchMatch? none = doc.FindFirst("xyz");
Console.WriteLine(none.HasValue); // false

FindNext — cursor-style forward scan

doc.Load("one two one three one");
int from = 4;  // after first "one"

SearchMatch? m = doc.FindNext("one", from);
Console.WriteLine(m?.Offset); // 8

m = doc.FindNext("one", m!.Value.Offset + 1);
Console.WriteLine(m?.Offset); // 18

FindNext — wrap-around (Ctrl+F behaviour)

int caretOffset = 18;
SearchMatch? hit = doc.FindNext("one", caretOffset + 1)
                ?? doc.FindNext("one", 0);   // wrap to beginning
Console.WriteLine(hit?.Offset); // 0  (wrapped)

FindPrev — backward scan

doc.Load("one two one three one");

SearchMatch? m = doc.FindPrev("one", doc.Length);
Console.WriteLine(m?.Offset); // 18

m = doc.FindPrev("one", m!.Value.Offset);
Console.WriteLine(m?.Offset); // 8

m = doc.FindPrev("one", m!.Value.Offset);
Console.WriteLine(m?.Offset); // 0

CountMatches

doc.Load("apple orange apple banana apple");
Console.WriteLine(doc.CountMatches("apple")); // 3
Console.WriteLine(doc.CountMatches("xyz"));   // 0

var opts = new SearchOptions { UseRegex = true };
Console.WriteLine(doc.CountMatches(@"\b\w{6}\b", opts)); // count 6-letter words

SearchMatch struct

doc.Load("Hello World");
SearchMatch? m = doc.FindFirst("World");

// Deconstruct offset and length
Console.WriteLine(m?.Offset);  // 6
Console.WriteLine(m?.Length);  // 5

// Extract the matched text
string text = doc.GetText(m!.Value.Offset, m.Value.Length);
Console.WriteLine(text); // "World"

Decorations — Every DecorationType

ErrorSquiggle

using TextAPI.Core.Decorations;

doc.Load("int x = unknownVar;");

// Mark "unknownVar" as an error
var m = doc.FindFirst("unknownVar")!.Value;
Guid id = doc.AddDecoration(m.Offset, m.Offset + m.Length,
    DecorationType.ErrorSquiggle, tag: "CS0103");

var d = doc.GetDecorationsInRange(0, doc.Length).First();
Console.WriteLine($"{d.Type} [{d.Start},{d.End}) tag={d.Tag}");
// ErrorSquiggle [8,18) tag=CS0103

WarningSquiggle

doc.Load("var x = 1;  // unused");
var m = doc.FindFirst("x")!.Value;
doc.AddDecoration(m.Offset, m.Offset + m.Length,
    DecorationType.WarningSquiggle, tag: "CS0219", data: "Variable 'x' assigned but never used");

InfoSquiggle

doc.AddDecoration(0, 10, DecorationType.InfoSquiggle, tag: "hint", data: "Consider using var");

SyntaxHighlight

// Token colouring — typically set by the tokeniser automatically
// but you can add manual highlights too
doc.AddDecoration(0, 3, DecorationType.SyntaxHighlight, tag: "keyword");

SearchMatch (highlight)

doc.Load("foo bar foo baz foo");

// Highlight all matches for the current search term
doc.Decorations.RemoveAllOfType(DecorationType.SearchMatch);
foreach (var m in doc.FindAll("foo"))
    doc.AddDecoration(m.Offset, m.Offset + m.Length,
        DecorationType.SearchMatch);

Console.WriteLine(doc.Decorations.Count); // 3

Bookmark decoration

Guid bookmarkId = doc.AddDecoration(0, 5, DecorationType.Bookmark, tag: "F1");

Selection

// Typically set/unset by the cursor, but you can layer it manually
doc.AddDecoration(0, 5, DecorationType.Selection);

Custom

record DiagnosticPayload(string Message, string Severity);

doc.AddDecoration(10, 20, DecorationType.Custom,
    tag: "myTool",
    data: new DiagnosticPayload("Something unusual here", "info"));

// Retrieve the payload
var d = doc.GetDecorationsInRange(0, doc.Length)
           .First(x => x.Tag == "myTool");
var payload = (DiagnosticPayload)d.Data!;
Console.WriteLine(payload.Message); // "Something unusual here"

RemoveDecoration

Guid id = doc.AddDecoration(0, 5, DecorationType.ErrorSquiggle, tag: "E1");
Console.WriteLine(doc.Decorations.Count); // 1

bool removed = doc.RemoveDecoration(id);
Console.WriteLine(removed);                 // true
Console.WriteLine(doc.Decorations.Count);  // 0

RemoveAllOfType

doc.AddDecoration(0, 5,  DecorationType.ErrorSquiggle);
doc.AddDecoration(6, 10, DecorationType.WarningSquiggle);
doc.AddDecoration(11, 15, DecorationType.ErrorSquiggle);

doc.Decorations.RemoveAllOfType(DecorationType.ErrorSquiggle);
Console.WriteLine(doc.Decorations.Count); // 1  (warning remains)

GetDecorationsInRange

doc.Load("Hello World");
doc.AddDecoration(0, 5,  DecorationType.ErrorSquiggle);
doc.AddDecoration(6, 11, DecorationType.Bookmark);
doc.AddDecoration(3, 8,  DecorationType.WarningSquiggle);

// All decorations overlapping range [2, 7)
var overlapping = doc.GetDecorationsInRange(2, 7).ToList();
foreach (var d in overlapping)
    Console.WriteLine($"{d.Type} [{d.Start},{d.End})");
// ErrorSquiggle [0,5)
// WarningSquiggle [3,8)

Decoration — automatic offset tracking

doc.Load("Hello World");
Guid id = doc.AddDecoration(6, 11, DecorationType.Bookmark); // on "World"

// Insert text before the decoration
doc.Insert(0, "XXX "); // +4 chars at offset 0

// Decoration automatically shifted
var d = doc.Decorations.GetDecorationsInRange(0, doc.Length).First(x => x.Id == id);
Console.WriteLine($"[{d.Start},{d.End})");  // [10,15)  (6+4=10, 11+4=15)

Decoration — deleted when text is removed

doc.Load("Hello World");
Guid id = doc.AddDecoration(6, 11, DecorationType.Bookmark);

// Delete text that completely covers the decoration
doc.Delete(5, 6);  // deletes " World"

// Decoration is removed automatically
var remaining = doc.Decorations.GetDecorationsInRange(0, doc.Length).ToList();
Console.WriteLine(remaining.Any(x => x.Id == id)); // false

TextDiff — Every Method

Diff(string, string) — basic line diff

using TextAPI.Core.Diff;

string oldText = "line1\nline2\nline3";
string newText = "line1\nLINE2 CHANGED\nline3\nline4";

DiffResult result = TextDiff.Diff(oldText, newText);

Console.WriteLine(result.HasChanges);     // true
Console.WriteLine(result.AddedLines);     // 2  (LINE2 CHANGED + line4)
Console.WriteLine(result.DeletedLines);   // 1  (line2)
Console.WriteLine(result.ToUnifiedDiff()); // unified diff string

Diff — identical texts

var r = TextDiff.Diff("same", "same");
Console.WriteLine(r.HasChanges);   // false
Console.WriteLine(r.AddedLines);   // 0
Console.WriteLine(r.DeletedLines); // 0

Diff with DiffOptions — IgnoreCase

var opts = new DiffOptions { IgnoreCase = true };
var r    = TextDiff.Diff("Hello", "hello", opts);
Console.WriteLine(r.HasChanges); // false  (case ignored)

Diff with DiffOptions — IgnoreWhitespace

var opts = new DiffOptions { IgnoreWhitespace = true };
var r    = TextDiff.Diff("  foo  bar  ", "foo bar", opts);
Console.WriteLine(r.HasChanges); // false  (whitespace ignored)

Diff with DiffOptions — MaxEditDistance

var opts = new DiffOptions { MaxEditDistance = 5 };
var r    = TextDiff.Diff(bigFile1, bigFile2, opts);
// if edit distance > 5, returns partial result and stops early

Diff(string[], string[]) — pre-split lines

string[] old = { "a", "b", "c" };
string[] neu = { "a", "X", "c", "d" };
var r = TextDiff.Diff(old, neu);
Console.WriteLine(r.AddedLines);   // 2 (X + d)
Console.WriteLine(r.DeletedLines); // 1 (b)

Diff(TextDocument, TextDocument)

var docA = new TextDocument(); docA.Load("version 1\ncommon");
var docB = new TextDocument(); docB.Load("version 2\ncommon");
var r = TextDiff.Diff(docA, docB);
Console.WriteLine(r.AddedLines); // 1

DiffResult.Hunks — iterate changes

var r = TextDiff.Diff("a\nb\nc\nd", "a\nX\nc\nY");

foreach (var hunk in r.Hunks)
{
    string sign = hunk.Kind switch
    {
        DiffKind.Insert => "+",
        DiffKind.Delete => "-",
        _               => " "
    };
    foreach (var line in hunk.Lines)
        Console.WriteLine($"{sign} {line}");
}
//   a
// - b
// + X
//   c
// - d
// + Y

DiffHunk — all properties

foreach (var hunk in r.Hunks.Where(h => h.Kind != DiffKind.Equal))
{
    Console.WriteLine($"Kind:     {hunk.Kind}");
    Console.WriteLine($"OldStart: {hunk.OldStart}, OldCount: {hunk.OldCount}");
    Console.WriteLine($"NewStart: {hunk.NewStart}, NewCount: {hunk.NewCount}");
    Console.WriteLine($"Lines:    {string.Join(", ", hunk.Lines)}");
}

ToUnifiedDiff — custom paths and context

var r = TextDiff.Diff(oldText, newText);

// Default (3 context lines)
Console.WriteLine(r.ToUnifiedDiff());

// Custom paths, 5 context lines
Console.WriteLine(r.ToUnifiedDiff(
    oldPath:      "v1/file.cs",
    newPath:      "v2/file.cs",
    contextLines: 5
));
// --- v1/file.cs
// +++ v2/file.cs
// @@ -1,5 +1,5 @@
//  context
// -old
// +new
//  context

DiffChars — inline character-level diff

var spans = TextDiff.DiffChars("the cat sat", "the bat sat");

foreach (var s in spans)
{
    string prefix = s.Kind switch
    {
        DiffKind.Insert => "[+]",
        DiffKind.Delete => "[-]",
        _               => "   "
    };
    Console.WriteLine($"{prefix}{s.Text}");
}
//    the
// [-]c
// [+]b
//    at sat

DiffChars — no changes

var spans = TextDiff.DiffChars("identical", "identical");
Console.WriteLine(spans.Count); // 1
Console.WriteLine(spans[0].Kind); // Equal

DiffChars for per-line inline rendering

// Apply line-level diff first, then char-level within changed lines
var lineDiff = TextDiff.Diff(oldText, newText);

var deletions = lineDiff.Hunks.Where(h => h.Kind == DiffKind.Delete);
var insertions = lineDiff.Hunks.Where(h => h.Kind == DiffKind.Insert);

foreach (var (del, ins) in deletions.Zip(insertions))
{
    var charDiff = TextDiff.DiffChars(del.Lines[0], ins.Lines[0]);
    foreach (var span in charDiff)
        Console.Write(span.Kind == DiffKind.Delete ? $"\x1b[31m{span.Text}\x1b[0m"
                      : span.Kind == DiffKind.Insert ? $"\x1b[32m{span.Text}\x1b[0m"
                      : span.Text);
    Console.WriteLine();
}
